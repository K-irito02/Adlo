# 文件系统操作实施计划

## 模块概述

**包名**: `@ai-digital-life/fs-ops`

**优先级**: P0 - AI 修改代码的基础能力

**目标**: 为 AI 数字生命提供对项目文件的完整读写能力，封装文件读写编辑、内容搜索、变更快照与回滚等关键能力，同时针对大文件处理与并发操作进行专项优化。

## 设计原则

- **行级精确编辑**: 代码修改场景需要精确到行的 insert/delete/replace
- **变更可追溯**: 每次修改前自动创建快照，支持多级回滚
- **大文件友好**: 流式读取、分片加载，避免内存溢出
- **并发安全**: 文件操作队列化，读写锁保护
- **安全边界**: 文件操作限制在工作区范围内，禁止越界访问

## Phase 1 职责边界

### 必须交付

- 文件读写：readFile、readFileLines、writeFile、deleteFile
- 行级编辑：insert、delete、replace，行号偏移自动修正
- Grep/Glob 搜索
- 变更快照与回滚
- 并发控制（操作队列化）
- 大文件分片读取

### 只做骨架

- AST 符号索引（Phase 1 仅做文件级操作，符号索引延后）
- 跨文件理解与依赖追踪
- 符号重命名
- 远程文件系统

```typescript
export class PathValidator {
  private allowedRoots: Set<string>;

  constructor(
    private readonly config: { workspaceRoots: string[] },
    private readonly logger: Logger
  ) {
    this.allowedRoots = new Set(
      config.workspaceRoots.map(r => normalizePath(r))
    );
  }

  validate(filePath: string): Result<string> {
    const normalized = normalizePath(filePath);

    if (normalized.includes('..')) {
      return Result.err(new SecurityError('Path traversal detected'));
    }

    const isWithinWorkspace = Array.from(this.allowedRoots).some(root =>
      normalized.startsWith(root)
    );

    if (!isWithinWorkspace) {
      return Result.err(new SecurityError('Path outside workspace boundaries'));
    }

    const sensitivePatterns = [
      /\/\.env/i,
      /\/\.ssh\//i,
      /\/credentials/i,
      /\/\.git\/config/i,
    ];

    for (const pattern of sensitivePatterns) {
      if (pattern.test(normalized)) {
        this.logger.warn('Access to sensitive path attempted', { path: normalized });
      }
    }

    return Result.ok(normalized);
  }

  addWorkspaceRoot(root: string): void {
    this.allowedRoots.add(normalizePath(root));
  }

  removeWorkspaceRoot(root: string): void {
    this.allowedRoots.delete(normalizePath(root));
  }
}
```

```typescript
export class FileReader {
  constructor(
    private readonly pathValidator: PathValidator,
    private readonly logger: Logger
  ) {}

  async readFile(filePath: string, options?: FileReadOptions): Promise<Result<string>> {
    const validation = this.pathValidator.validate(filePath);
    if (Result.isErr(validation)) return validation;

    try {
      const stat = await fs.stat(filePath);
      if (stat.size > 10 * 1024 * 1024) {
        this.logger.warn('Large file read, using streaming', {
          path: filePath,
          size: stat.size,
        });
        return this.readLargeFile(filePath, options);
      }

      const content = await fs.readFile(filePath, options?.encoding || 'utf-8');

      if (options?.startLine || options?.endLine) {
        return Result.ok(this.extractLines(content, options.startLine, options.endLine));
      }

      return Result.ok(content);
    } catch (error) {
      return Result.err(new FileSystemError(`Failed to read file: ${filePath}`, error as Error));
    }
  }

  async readFileLines(
    filePath: string,
    startLine: number,
    endLine: number
  ): Promise<Result<string>> {
    return this.readFile(filePath, { startLine, endLine });
  }

  async getFileInfo(filePath: string): Promise<Result<FileInfo>> {
    const validation = this.pathValidator.validate(filePath);
    if (Result.isErr(validation)) return validation as Result<FileInfo>;

    try {
      const stat = await fs.stat(filePath);
      return Result.ok({
        path: filePath,
        size: stat.size,
        modifiedTime: stat.mtimeMs,
        isDirectory: stat.isDirectory(),
      });
    } catch (error) {
      return Result.err(new FileSystemError(`Failed to get file info: ${filePath}`, error as Error));
    }
  }

  private extractLines(content: string, startLine?: number, endLine?: number): string {
    const lines = content.split('\n');
    const start = (startLine || 1) - 1;
    const end = endLine || lines.length;
    return lines.slice(start, end).join('\n');
  }

  private async readLargeFile(filePath: string, options?: FileReadOptions): Promise<Result<string>> {
    const stream = createReadStream(filePath, { encoding: options?.encoding || 'utf-8' });
    const chunks: string[] = [];
    let lineCount = 0;
    const startLine = options?.startLine || 1;
    const endLine = options?.endLine || Infinity;

    for await (const chunk of stream) {
      const lines = chunk.split('\n');
      for (const line of lines) {
        lineCount++;
        if (lineCount >= startLine && lineCount <= endLine) {
          chunks.push(line);
        }
        if (lineCount > endLine) {
          stream.destroy();
          break;
        }
      }
    }

    return Result.ok(chunks.join('\n'));
  }
}

export class FileWriter {
  constructor(
    private readonly pathValidator: PathValidator,
    private readonly snapshotManager: SnapshotManager,
    private readonly logger: Logger
  ) {}

  async writeFile(
    filePath: string,
    content: string,
    options?: FileWriteOptions
  ): Promise<Result<void>> {
    const validation = this.pathValidator.validate(filePath);
    if (Result.isErr(validation)) return validation as Result<void>;

    try {
      if (options?.createDir) {
        const dir = path.dirname(filePath);
        await fs.mkdir(dir, { recursive: true });
      }

      const exists = await this.fileExists(filePath);
      if (exists && !options?.overwrite) {
        return Result.err(new FileSystemError(`File already exists: ${filePath}`));
      }

      if (exists) {
        await this.snapshotManager.createSnapshot(filePath, 'Before write');
      }

      await fs.writeFile(filePath, content, options?.encoding || 'utf-8');
      return Result.ok(undefined as void);
    } catch (error) {
      return Result.err(new FileSystemError(`Failed to write file: ${filePath}`, error as Error));
    }
  }

  async deleteFile(filePath: string): Promise<Result<void>> {
    const validation = this.pathValidator.validate(filePath);
    if (Result.isErr(validation)) return validation as Result<void>;

    try {
      const exists = await this.fileExists(filePath);
      if (!exists) {
        return Result.err(new FileSystemError(`File not found: ${filePath}`));
      }

      await this.snapshotManager.createSnapshot(filePath, 'Before delete');
      await fs.unlink(filePath);
      return Result.ok(undefined as void);
    } catch (error) {
      return Result.err(new FileSystemError(`Failed to delete file: ${filePath}`, error as Error));
    }
  }

  private async fileExists(filePath: string): Promise<boolean> {
    try {
      await fs.access(filePath);
      return true;
    } catch {
      return false;
    }
  }
}
```

```typescript
export class LineEditor {
  constructor(
    private readonly fileReader: FileReader,
    private readonly fileWriter: FileWriter,
    private readonly logger: Logger
  ) {}

  async applyEdits(
    filePath: string,
    operations: EditOperation[]
  ): Promise<Result<EditResult>> {
    const contentResult = await this.fileReader.readFile(filePath);
    if (Result.isErr(contentResult)) return contentResult as Result<EditResult>;

    const originalContent = contentResult.value;
    const originalLines = originalContent.split('\n');
    const lineCountBefore = originalLines.length;

    const sortedOps = [...operations].sort((a, b) => {
      const lineA = 'line' in a ? a.line : a.startLine;
      const lineB = 'line' in b ? b.line : b.startLine;
      return lineB - lineA;
    });

    let currentLines = [...originalLines];

    for (const op of sortedOps) {
      const result = this.applyOperation(currentLines, op);
      if (Result.isErr(result)) return result as Result<EditResult>;
      currentLines = result.value;
    }

    const newContent = currentLines.join('\n');
    const writeResult = await this.fileWriter.writeFile(filePath, newContent, { overwrite: true });

    if (Result.isErr(writeResult)) return writeResult as Result<EditResult>;

    return Result.ok({
      filePath,
      applied: operations,
      snapshotId: '',
      lineCountBefore,
      lineCountAfter: currentLines.length,
    });
  }

  private applyOperation(lines: string[], op: EditOperation): Result<string[]> {
    switch (op.type) {
      case 'insert':
        return this.insertLines(lines, op.line, op.content);

      case 'delete':
        return this.deleteLines(lines, op.startLine, op.endLine);

      case 'replace':
        return this.replaceLines(lines, op.startLine, op.endLine, op.content);

      default:
        return Result.err(new ValidationError(`Unknown edit operation: ${(op as EditOperation).type}`));
    }
  }

  private insertLines(lines: string[], line: number, content: string): Result<string[]> {
    if (line < 1 || line > lines.length + 1) {
      return Result.err(new ValidationError(`Invalid line number: ${line}`));
    }

    const newLines = content.split('\n');
    const result = [
      ...lines.slice(0, line - 1),
      ...newLines,
      ...lines.slice(line - 1),
    ];
    return Result.ok(result);
  }

  private deleteLines(lines: string[], startLine: number, endLine: number): Result<string[]> {
    if (startLine < 1 || endLine > lines.length || startLine > endLine) {
      return Result.err(new ValidationError(`Invalid line range: ${startLine}-${endLine}`));
    }

    const result = [
      ...lines.slice(0, startLine - 1),
      ...lines.slice(endLine),
    ];
    return Result.ok(result);
  }

  private replaceLines(
    lines: string[],
    startLine: number,
    endLine: number,
    content: string
  ): Result<string[]> {
    if (startLine < 1 || endLine > lines.length || startLine > endLine) {
      return Result.err(new ValidationError(`Invalid line range: ${startLine}-${endLine}`));
    }

    const newLines = content.split('\n');
    const result = [
      ...lines.slice(0, startLine - 1),
      ...newLines,
      ...lines.slice(endLine),
    ];
    return Result.ok(result);
  }
}
```

```typescript
export class GrepSearch {
  constructor(
    private readonly pathValidator: PathValidator,
    private readonly logger: Logger
  ) {}

  async search(options: GrepOptions, workspaceRoot: string): Promise<GrepMatch[]> {
    const files = await this.collectFiles(workspaceRoot, options.filePattern, options.excludePatterns);
    const results: GrepMatch[] = [];
    const maxResults = options.maxResults || 100;
    const pattern = typeof options.pattern === 'string'
      ? new RegExp(options.pattern, 'g')
      : options.pattern;
    const contextLines = options.contextLines || 0;

    for (const file of files) {
      if (results.length >= maxResults) break;

      try {
        const content = await fs.readFile(file, 'utf-8');
        const lines = content.split('\n');

        for (let i = 0; i < lines.length; i++) {
          if (results.length >= maxResults) break;

          pattern.lastIndex = 0;
          if (pattern.test(lines[i])) {
            results.push({
              filePath: file,
              lineNumber: i + 1,
              lineContent: lines[i],
              contextBefore: lines.slice(Math.max(0, i - contextLines), i),
              contextAfter: lines.slice(i + 1, i + 1 + contextLines),
            });
          }
        }
      } catch {
        this.logger.debug('Failed to search in file', { file });
      }
    }

    return results;
  }

  private async collectFiles(
    root: string,
    filePattern?: string,
    excludePatterns?: string[]
  ): Promise<string[]> {
    const globber = new GlobSearch();
    const pattern = filePattern || '**/*';
    return globber.search({ pattern, excludePatterns, sortBy: 'path' }, root);
  }
}

export class GlobSearch {
  async search(options: GlobOptions, workspaceRoot: string): Promise<string[]> {
    const maxResults = options.maxResults || 1000;
    const results: string[] = [];
    const exclude = new Set(options.excludePatterns || [
      'node_modules/**',
      '.git/**',
      'dist/**',
      'build/**',
    ]);

    const globPattern = new Glob(options.pattern, {
      cwd: workspaceRoot,
      ignore: Array.from(exclude),
      nodir: true,
    });

    for await (const file of globPattern) {
      results.push(path.resolve(workspaceRoot, file));
      if (results.length >= maxResults) break;
    }

    if (options.sortBy === 'modifiedTime') {
      const withStats = await Promise.all(
        results.map(async f => ({
          path: f,
          mtime: (await fs.stat(f)).mtimeMs,
        }))
      );
      withStats.sort((a, b) => b.mtime - a.mtime);
      return withStats.map(f => f.path);
    }

    return results.sort();
  }
}
```

```typescript
export class SnapshotManager {
  private store: SnapshotStore;

  constructor(
    private readonly config: { maxSnapshots: number },
    private readonly logger: Logger
  ) {
    this.store = new SnapshotStore(config.maxSnapshots || 50);
  }

  async createSnapshot(filePath: string, description: string): Promise<FileSnapshot> {
    const content = await fs.readFile(filePath, 'utf-8');
    const snapshot: FileSnapshot = {
      id: this.generateId(),
      filePath,
      content,
      timestamp: Date.now(),
      description,
    };

    this.store.add(snapshot);
    this.logger.debug('Snapshot created', {
      snapshotId: snapshot.id,
      filePath,
      description,
    });

    return snapshot;
  }

  async restoreSnapshot(snapshotId: string): Promise<Result<void>> {
    const snapshot = this.store.get(snapshotId);
    if (!snapshot) {
      return Result.err(new SnapshotNotFoundError(snapshotId));
    }

    try {
      await fs.writeFile(snapshot.filePath, snapshot.content, 'utf-8');
      this.logger.info('Snapshot restored', {
        snapshotId,
        filePath: snapshot.filePath,
      });
      return Result.ok(undefined as void);
    } catch (error) {
      return Result.err(new FileSystemError('Failed to restore snapshot', error as Error));
    }
  }

  getSnapshots(filePath?: string): FileSnapshot[] {
    if (filePath) {
      return this.store.getByFilePath(filePath);
    }
    return this.store.getAll();
  }

  private generateId(): string {
    return `snap-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
  }
}

export class SnapshotStore {
  private snapshots: Map<string, FileSnapshot> = new Map();
  private fileIndex: Map<string, string[]> = new Map();

  constructor(private readonly maxSize: number) {}

  add(snapshot: FileSnapshot): void {
    this.snapshots.set(snapshot.id, snapshot);

    if (!this.fileIndex.has(snapshot.filePath)) {
      this.fileIndex.set(snapshot.filePath, []);
    }
    this.fileIndex.get(snapshot.filePath)!.push(snapshot.id);

    if (this.snapshots.size > this.maxSize) {
      this.evictOldest();
    }
  }

  get(id: string): FileSnapshot | undefined {
    return this.snapshots.get(id);
  }

  getByFilePath(filePath: string): FileSnapshot[] {
    const ids = this.fileIndex.get(filePath) || [];
    return ids
      .map(id => this.snapshots.get(id))
      .filter((s): s is FileSnapshot => s !== undefined)
      .sort((a, b) => b.timestamp - a.timestamp);
  }

  getAll(): FileSnapshot[] {
    return Array.from(this.snapshots.values())
      .sort((a, b) => b.timestamp - a.timestamp);
  }

  private evictOldest(): void {
    let oldest: FileSnapshot | null = null;
    for (const snapshot of this.snapshots.values()) {
      if (!oldest || snapshot.timestamp < oldest.timestamp) {
        oldest = snapshot;
      }
    }
    if (oldest) {
      this.snapshots.delete(oldest.id);
      const ids = this.fileIndex.get(oldest.filePath);
      if (ids) {
        const idx = ids.indexOf(oldest.id);
        if (idx >= 0) ids.splice(idx, 1);
        if (ids.length === 0) this.fileIndex.delete(oldest.filePath);
      }
    }
  }
}
```

### 任务6: 并发控制 (2天)

```typescript
export class OperationQueue {
  private queue: Array<{
    operation: () => Promise<unknown>;
    resolve: (value: unknown) => void;
    reject: (reason: unknown) => void;
  }> = [];
  private running = 0;
  private maxConcurrency: number;

  constructor(maxConcurrency = 4) {
    this.maxConcurrency = maxConcurrency;
  }

  async enqueue<T>(operation: () => Promise<T>): Promise<T> {
    return new Promise<T>((resolve, reject) => {
      this.queue.push({ operation, resolve, reject });
      this.process();
    });
  }

  private process(): void {
    while (this.running < this.maxConcurrency && this.queue.length > 0) {
      const item = this.queue.shift()!;
      this.running++;

      item.operation()
        .then(result => {
          item.resolve(result);
        })
        .catch(error => {
          item.reject(error);
        })
        .finally(() => {
          this.running--;
          this.process();
        });
    }
  }
}

export class ReadWriteLock {
  private readers = 0;
  private writer = false;
  private waitingReaders: Array<() => void> = [];
  private waitingWriters: Array<() => void> = [];

  async readLock(): Promise<Disposable> {
    if (this.writer) {
      await new Promise<void>(resolve => this.waitingReaders.push(resolve));
    }
    this.readers++;
    return {
      dispose: () => this.releaseRead(),
    };
  }

  async writeLock(): Promise<Disposable> {
    if (this.writer || this.readers > 0) {
      await new Promise<void>(resolve => this.waitingWriters.push(resolve));
    }
    this.writer = true;
    return {
      dispose: () => this.releaseWrite(),
    };
  }

  private releaseRead(): void {
    this.readers--;
    if (this.readers === 0 && this.waitingWriters.length > 0) {
      const next = this.waitingWriters.shift()!;
      next();
    }
  }

  private releaseWrite(): void {
    this.writer = false;
    if (this.waitingWriters.length > 0) {
      const next = this.waitingWriters.shift()!;
      next();
    } else {
      while (this.waitingReaders.length > 0) {
        const next = this.waitingReaders.shift()!;
        next();
      }
    }
  }
}

export interface Disposable {
  dispose: () => void;
}
```

```typescript
    await manager.createSnapshot('/test/file1.ts', '1');
    await manager.createSnapshot('/test/file2.ts', '2');
    await manager.createSnapshot('/test/file3.ts', '3');
```
