# 终端命令执行实施计划

## 模块概述

**包名**: `@ai-digital-life/terminal`

**优先级**: P0 - AI 验证代码正确性的关键能力

**目标**: 为 AI 数字生命提供安全的命令行操作能力，通过白名单机制、确认模式和环境隔离等多层安全策略，确保命令执行过程可控、可审计、可回溯。

## 设计原则

- **安全默认开启**: 危险命令默认拦截，安全命令才自动执行
- **多层防御**: 命令分类 → 确认模式 → 路径限制 → 审批流程
- **输出结构化**: 编译错误、测试结果等可解析为结构化数据
- **资源可控**: 超时控制、输出限制、进程生命周期管理
- **可审计**: 每次命令执行都有完整记录

## Phase 1 职责边界

### 必须交付

- 基础命令执行（Shell 执行 + 直接执行）
- 输出流实时捕获（stdout/stderr 分离）
- 安全控制（命令分类 + 确认模式 + 白名单）
- 超时控制与手动终止
- 命令历史记录
- 基础输出解析（编译错误、测试结果）

### 只做骨架

- 交互式终端模拟器
- 远程 SSH 执行
- Docker 容器隔离
- 命令 DAG 编排
- 环境变量深度隔离

### 任务1: 命令分类与风险评估 (3天)

```typescript
export class CommandClassifier {
  private rules: ClassificationRule[];

  constructor() {
    this.rules = this.buildDefaultRules();
  }

  classify(command: string, args: string[] = []): CommandRiskLevel {
    const fullCommand = `${command} ${args.join(' ')}`.trim().toLowerCase();

    for (const rule of this.rules) {
      if (rule.pattern.test(fullCommand)) {
        return rule.riskLevel;
      }
    }

    return 'medium';
  }

  private buildDefaultRules(): ClassificationRule[] {
    return [
      { pattern: /^rm\s+(-rf?|-fr?)\s+\//, riskLevel: 'blocked' },
      { pattern: /^format\s+[a-z]:/i, riskLevel: 'blocked' },
      { pattern: /^dd\s+if=/, riskLevel: 'blocked' },
      { pattern: /^sudo\s+rm/, riskLevel: 'blocked' },
      { pattern: /^:\(\)\{\s*:\|:\&\s*\}\s*;/, riskLevel: 'blocked' },

      { pattern: /^rm\s+/, riskLevel: 'high' },
      { pattern: /^sudo\s+/, riskLevel: 'high' },
      { pattern: /^chmod\s+777/, riskLevel: 'high' },
      { pattern: /^git\s+(push\s+--force|reset\s+--hard)/, riskLevel: 'high' },

      { pattern: /^npm\s+run/, riskLevel: 'medium' },
      { pattern: /^python\s+/, riskLevel: 'medium' },
      { pattern: /^cargo\s+run/, riskLevel: 'medium' },
      { pattern: /^go\s+run/, riskLevel: 'medium' },
      { pattern: /^git\s+(commit|push|merge)/, riskLevel: 'medium' },

      { pattern: /^npm\s+install/, riskLevel: 'low' },
      { pattern: /^pip\s+install/, riskLevel: 'low' },
      { pattern: /^go\s+(build|mod)/, riskLevel: 'low' },
      { pattern: /^cargo\s+build/, riskLevel: 'low' },

      { pattern: /^ls\b/, riskLevel: 'safe' },
      { pattern: /^cat\b/, riskLevel: 'safe' },
      { pattern: /^git\s+(status|log|diff|branch)/, riskLevel: 'safe' },
      { pattern: /^node\s+-v/, riskLevel: 'safe' },
      { pattern: /^npm\s+(-v|--version)/, riskLevel: 'safe' },
      { pattern: /^echo\b/, riskLevel: 'safe' },
      { pattern: /^pwd\b/, riskLevel: 'safe' },
      { pattern: /^which\b/, riskLevel: 'safe' },
      { pattern: /^npm\s+test/, riskLevel: 'safe' },
      { pattern: /^npx\s+vitest/, riskLevel: 'safe' },
    ];
  }
}

export interface ClassificationRule {
  pattern: RegExp;
  riskLevel: CommandRiskLevel;
}
```

### 任务2: 审批管理 (3天)

```typescript
export class ApprovalManager {
  private sessionAllowed: Set<string> = new Set();
  private alwaysAllowed: Set<string> = new Set();
  private pendingApprovals: Map<string, {
    request: ApprovalRequest;
    resolve: (decision: ApprovalDecision) => void;
  }> = new Map();

  constructor(
    private readonly classifier: CommandClassifier,
    private readonly config: { mode: ConfirmationMode },
    private readonly storage: Storage,
    private readonly logger: Logger
  ) {}

  async requestApproval(commandRequest: CommandRequest): Promise<Result<ApprovalDecision>> {
    const riskLevel = this.classifier.classify(commandRequest.command, commandRequest.args);

    if (riskLevel === 'blocked') {
      this.logger.warn('Blocked command attempted', {
        command: commandRequest.command,
        source: commandRequest.metadata?.source,
      });
      return Result.ok('reject');
    }

    if (this.config.mode === 'auto' && riskLevel === 'safe') {
      return Result.ok('allow-once');
    }

    if (this.config.mode === 'readonly' && riskLevel !== 'safe') {
      return Result.ok('reject');
    }

    if (this.isAlwaysAllowed(commandRequest.command)) {
      return Result.ok('always-allow');
    }

    if (this.isSessionAllowed(commandRequest.command)) {
      return Result.ok('allow-session');
    }

    if (this.config.mode === 'smart') {
      if (riskLevel === 'safe' || riskLevel === 'low') {
        return Result.ok('allow-once');
      }
    }

    const request: ApprovalRequest = {
      commandId: this.generateId(),
      command: `${commandRequest.command} ${commandRequest.args?.join(' ') || ''}`.trim(),
      riskLevel,
      reason: this.getRiskReason(riskLevel),
      impact: this.assessImpact(commandRequest),
    };

    return new Promise<Result<ApprovalDecision>>((resolve) => {
      this.pendingApprovals.set(request.commandId, { request, resolve });
      this.emitApprovalRequest(request);
    });
  }

  resolveApproval(commandId: string, decision: ApprovalDecision): void {
    const pending = this.pendingApprovals.get(commandId);
    if (!pending) return;

    this.pendingApprovals.delete(commandId);

    if (decision === 'allow-session') {
      this.sessionAllowed.add(pending.request.command);
    } else if (decision === 'always-allow') {
      this.alwaysAllowed.add(pending.request.command);
      this.persistAlwaysAllowed();
    }

    this.recordDecision(pending.request, decision);
    pending.resolve(decision);
  }

  private isAlwaysAllowed(command: string): boolean {
    return this.alwaysAllowed.has(command);
  }

  private isSessionAllowed(command: string): boolean {
    return this.sessionAllowed.has(command);
  }

  private getRiskReason(riskLevel: CommandRiskLevel): string {
    const reasons: Record<CommandRiskLevel, string> = {
      safe: '该命令被认为是安全的',
      low: '该命令风险较低',
      medium: '该命令可能修改项目文件或执行代码',
      high: '该命令具有破坏性风险',
      blocked: '该命令被禁止执行',
    };
    return reasons[riskLevel];
  }

  private assessImpact(request: CommandRequest): string[] {
    const impacts: string[] = [];
    if (request.cwd) impacts.push(`工作目录: ${request.cwd}`);
    if (/install/i.test(request.command)) impacts.push('可能安装新依赖');
    if (/rm|del|remove/i.test(request.command)) impacts.push('可能删除文件');
    if (/git\s+push/i.test(request.command)) impacts.push('可能推送到远程仓库');
    return impacts;
  }

  private generateId(): string {
    return `cmd-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
  }
}
```

```typescript
export class CommandExecutor {
  private processManager: ProcessManager;
  private outputLimit = 10000;

  constructor(
    private readonly approvalManager: ApprovalManager,
    private readonly classifier: CommandClassifier,
    private readonly logger: Logger
  ) {
    this.processManager = new ProcessManager();
  }

  async execute(request: CommandRequest): Promise<CommandResult> {
    const riskLevel = this.classifier.classify(request.command, request.args);

    if (riskLevel === 'blocked') {
      return {
        id: this.generateId(),
        command: request.command,
        exitCode: -1,
        stdout: '',
        stderr: 'Command blocked: this command is not allowed',
        riskLevel,
        startTime: Date.now(),
        endTime: Date.now(),
        timedOut: false,
        killed: false,
      };
    }

    const approval = await this.approvalManager.requestApproval(request);
    if (Result.isErr(approval) || approval.value === 'reject') {
      return {
        id: this.generateId(),
        command: request.command,
        exitCode: -1,
        stdout: '',
        stderr: 'Command rejected by user',
        riskLevel,
        startTime: Date.now(),
        endTime: Date.now(),
        timedOut: false,
        killed: false,
      };
    }

    const startTime = Date.now();
    const id = this.generateId();
    const timeout = request.timeout || 60000;

    try {
      const result = await this.spawnProcess(id, request, timeout);
      return {
        ...result,
        id,
        riskLevel,
        startTime,
        endTime: Date.now(),
        timedOut: false,
        killed: false,
      };
    } catch (error) {
      if (error instanceof TimeoutError) {
        return {
          id,
          command: request.command,
          exitCode: null,
          stdout: '',
          stderr: `Command timed out after ${timeout}ms`,
          riskLevel,
          startTime,
          endTime: Date.now(),
          timedOut: true,
          killed: true,
        };
      }

      return {
        id,
        command: request.command,
        exitCode: -1,
        stdout: '',
        stderr: (error as Error).message,
        riskLevel,
        startTime,
        endTime: Date.now(),
        timedOut: false,
        killed: false,
      };
    }
  }

  async executeStreaming(
    request: CommandRequest,
    onOutput: (output: CommandOutput) => void
  ): Promise<CommandResult> {
    const riskLevel = this.classifier.classify(request.command, request.args);

    if (riskLevel === 'blocked') {
      onOutput({
        stream: 'stderr',
        data: 'Command blocked',
        timestamp: Date.now(),
      });
      return this.createBlockedResult(request, riskLevel);
    }

    const approval = await this.approvalManager.requestApproval(request);
    if (Result.isErr(approval) || approval.value === 'reject') {
      onOutput({
        stream: 'stderr',
        data: 'Command rejected by user',
        timestamp: Date.now(),
      });
      return this.createRejectedResult(request, riskLevel);
    }

    const startTime = Date.now();
    const id = this.generateId();
    const timeout = request.timeout || 60000;

    const child = spawn(request.command, request.args || [], {
      cwd: request.cwd,
      env: { ...process.env, ...request.env },
      shell: request.shell ?? true,
    });

    this.processManager.register(id, child);

    let stdout = '';
    let stderr = '';
    let stdoutLines = 0;
    let stderrLines = 0;

    child.stdout?.on('data', (data: Buffer) => {
      const text = data.toString();
      stdoutLines += text.split('\n').length;
      if (stdoutLines <= this.outputLimit) {
        stdout += text;
        onOutput({ stream: 'stdout', data: text, timestamp: Date.now() });
      }
    });

    child.stderr?.on('data', (data: Buffer) => {
      const text = data.toString();
      stderrLines += text.split('\n').length;
      if (stderrLines <= this.outputLimit) {
        stderr += text;
        onOutput({ stream: 'stderr', data: text, timestamp: Date.now() });
      }
    });

    if (request.stdin) {
      child.stdin?.write(request.stdin);
      child.stdin?.end();
    }

    const timeoutHandle = setTimeout(() => {
      this.processManager.kill(id);
    }, timeout);

    return new Promise((resolve) => {
      child.on('close', (code) => {
        clearTimeout(timeoutHandle);
        this.processManager.unregister(id);
        resolve({
          id,
          command: request.command,
          exitCode: code,
          stdout,
          stderr,
          riskLevel,
          startTime,
          endTime: Date.now(),
          timedOut: false,
          killed: code === null,
        });
      });

      child.on('error', (error) => {
        clearTimeout(timeoutHandle);
        this.processManager.unregister(id);
        resolve({
          id,
          command: request.command,
          exitCode: -1,
          stdout,
          stderr: error.message,
          riskLevel,
          startTime,
          endTime: Date.now(),
          timedOut: false,
          killed: false,
        });
      });
    });
  }

  kill(commandId: string): boolean {
    return this.processManager.kill(commandId);
  }

  private generateId(): string {
    return `exec-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
  }

  private createBlockedResult(request: CommandRequest, riskLevel: CommandRiskLevel): CommandResult {
    return {
      id: this.generateId(),
      command: request.command,
      exitCode: -1,
      stdout: '',
      stderr: 'Command blocked',
      riskLevel,
      startTime: Date.now(),
      endTime: Date.now(),
      timedOut: false,
      killed: false,
    };
  }

  private createRejectedResult(request: CommandRequest, riskLevel: CommandRiskLevel): CommandResult {
    return {
      id: this.generateId(),
      command: request.command,
      exitCode: -1,
      stdout: '',
      stderr: 'Command rejected by user',
      riskLevel,
      startTime: Date.now(),
      endTime: Date.now(),
      timedOut: false,
      killed: false,
    };
  }
}
```

```typescript
export class OutputParser {
  parseCompilerErrors(output: string): ParsedError[] {
    const errors: ParsedError[] = [];
    const patterns = [
      /(.+):(\d+):(\d+):\s+error:\s+(.+)/g,
      /(.+):(\d+):\s+error:\s+(.+)/g,
      /Error:\s+(.+)\n\s+at\s+(.+):(\d+):(\d+)/g,
    ];

    for (const pattern of patterns) {
      let match;
      while ((match = pattern.exec(output)) !== null) {
        errors.push({
          filePath: match[1],
          lineNumber: parseInt(match[2], 10),
          columnNumber: match[3] ? parseInt(match[3], 10) : undefined,
          severity: 'error',
          message: match[4] || match[3] || match[1],
          source: 'compiler',
        });
      }
    }

    return errors;
  }

  parseTestResult(output: string): ParsedTestResult | null {
    const vitestPattern = /Tests\s+(\d+)\s+passed.*?(?:(\d+)\s+failed)?/;
    const vitestMatch = vitestPattern.exec(output);
    if (vitestMatch) {
      const passed = parseInt(vitestMatch[1], 10);
      const failed = parseInt(vitestMatch[2] || '0', 10);
      return {
        total: passed + failed,
        passed,
        failed,
        skipped: 0,
        failures: this.extractTestFailures(output),
      };
    }

    const jestPattern = /Tests:\s+(\d+)\s+failed,\s+(\d+)\s+passed,\s+(\d+)\s+total/;
    const jestMatch = jestPattern.exec(output);
    if (jestMatch) {
      return {
        total: parseInt(jestMatch[3], 10),
        passed: parseInt(jestMatch[2], 10),
        failed: parseInt(jestMatch[1], 10),
        skipped: 0,
        failures: this.extractTestFailures(output),
      };
    }

    return null;
  }

  private extractTestFailures(output: string): Array<{ testName: string; error: string; filePath?: string }> {
    const failures: Array<{ testName: string; error: string; filePath?: string }> = [];
    const pattern = /FAIL\s+(.+)\n\s*✕\s+(.+)\n([\s\S]*?)(?=FAIL|✕|Tests|$)/g;

    let match;
    while ((match = pattern.exec(output)) !== null) {
      failures.push({
        filePath: match[1],
        testName: match[2],
        error: match[3].trim(),
      });
    }

    return failures;
  }
}
```

```typescript
export class ProcessManager {
  private processes: Map<string, ChildProcess> = new Map();

  register(id: string, process: ChildProcess): void {
    this.processes.set(id, process);

    process.on('exit', () => {
      this.processes.delete(id);
    });
  }

  unregister(id: string): void {
    this.processes.delete(id);
  }

  kill(id: string): boolean {
    const process = this.processes.get(id);
    if (!process) return false;

    try {
      process.kill('SIGTERM');
      setTimeout(() => {
        if (this.processes.has(id)) {
          process.kill('SIGKILL');
        }
      }, 5000);
      return true;
    } catch {
      return false;
    }
  }

  killAll(): void {
    for (const [id] of this.processes) {
      this.kill(id);
    }
  }

  getRunningCount(): number {
    return this.processes.size;
  }
}
```

```typescript
export class CommandHistory {
  private entries: CommandResult[] = [];
  private maxEntries = 1000;

  constructor(private readonly storage: Storage) {}

  add(result: CommandResult): void {
    this.entries.push(result);
    if (this.entries.length > this.maxEntries) {
      this.entries = this.entries.slice(-this.maxEntries);
    }
  }

  query(options?: {
    command?: string;
    exitCode?: number;
    since?: number;
    limit?: number;
  }): CommandResult[] {
    let results = [...this.entries];

    if (options?.command) {
      const pattern = new RegExp(options.command, 'i');
      results = results.filter(r => pattern.test(r.command));
    }
    if (options?.exitCode !== undefined) {
      results = results.filter(r => r.exitCode === options.exitCode);
    }
    if (options?.since) {
      results = results.filter(r => r.startTime >= options.since!);
    }

    results.sort((a, b) => b.startTime - a.startTime);

    if (options?.limit) {
      results = results.slice(0, options.limit);
    }

    return results;
  }

  async persist(): Promise<void> {
    await this.storage.set('terminal:history', this.entries);
  }

  async restore(): Promise<void> {
    const stored = await this.storage.get<CommandResult[]>('terminal:history');
    if (stored) {
      this.entries = stored;
    }
  }
}
```
