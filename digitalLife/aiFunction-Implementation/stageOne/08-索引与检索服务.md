# 索引与检索服务实施计划

## 模块概述

**包名**: `@ai-digital-life/retrieval`

**优先级**: P0 - AI理解代码库的基础能力

**目标**: 把代码库转成AI可消费的结构化上下文，包括文件树、文本搜索、符号索引、仓库地图和相关性排序。

## 设计原则

- 让AI在中大型仓库中稳定工作
- 让上下文装配可解释、可控、可缓存
- 让"找到信息"和"把信息送进模型"成为明确的独立能力

```typescript
// src/indexer/file-tree.ts

import fs from 'fs/promises';
import path from 'path';
import { createLogger } from '@ai-digital-life/logger';
import type { FileEntry, FileInfo, IndexMetadata, IndexStatus } from '../types';

const logger = createLogger('file-tree-indexer');

const IGNORE_PATTERNS = [
  'node_modules',
  '.git',
  '.svn',
  '.hg',
  'dist',
  'build',
  'out',
  '.next',
  '.nuxt',
  'coverage',
  '.cache',
  '__pycache__',
  '.pytest_cache',
  'venv',
  '.venv',
  'env',
  '.env',
];

const BINARY_EXTENSIONS = [
  '.png', '.jpg', '.jpeg', '.gif', '.ico', '.svg',
  '.pdf', '.zip', '.tar', '.gz', '.rar',
  '.exe', '.dll', '.so', '.dylib',
  '.mp3', '.mp4', '.wav', '.avi',
];

export class FileTreeIndexer {
  private files = new Map<string, FileEntry>();
  private metadata: IndexMetadata | null = null;
  private ignorePatterns: RegExp[];
  private maxFileSize: number;

  constructor(options: {
    ignorePatterns?: string[];
    maxFileSize?: number;
  } = {}) {
    this.ignorePatterns = [
      ...IGNORE_PATTERNS.map(p => new RegExp(`(^|/)${p}(/|$)`)),
      ...(options.ignorePatterns ?? []),
    ];
    this.maxFileSize = options.maxFileSize ?? 1024 * 1024; // 1MB
  }

  async index(rootPath: string, workspaceId: string): Promise<IndexMetadata> {
    logger.info('Starting file tree indexing', { rootPath });
    
    this.files.clear();
    
    await this.walkDirectory(rootPath, rootPath);
    
    this.metadata = {
      workspaceId,
      version: this.generateVersion(),
      createdAt: Date.now(),
      updatedAt: Date.now(),
      fileCount: this.files.size,
      symbolCount: 0,
      status: 'ready',
    };
    
    logger.info('File tree indexing complete', {
      fileCount: this.files.size,
    });
    
    return this.metadata;
  }

  private async walkDirectory(rootPath: string, currentPath: string): Promise<void> {
    const entries = await fs.readdir(currentPath, { withFileTypes: true });
    
    for (const entry of entries) {
      const fullPath = path.join(currentPath, entry.name);
      const relativePath = path.relative(rootPath, fullPath);
      
      if (this.shouldIgnore(relativePath)) {
        continue;
      }
      
      if (entry.isDirectory()) {
        await this.walkDirectory(rootPath, fullPath);
      } else if (entry.isFile()) {
        await this.indexFile(fullPath, relativePath);
      }
    }
  }

  private async indexFile(fullPath: string, relativePath: string): Promise<void> {
    try {
      const stats = await fs.stat(fullPath);
      
      if (stats.size > this.maxFileSize) {
        return;
      }
      
      const ext = path.extname(fullPath).toLowerCase();
      
      if (BINARY_EXTENSIONS.includes(ext)) {
        return;
      }
      
      const entry: FileEntry = {
        path: relativePath,
        name: path.basename(fullPath),
        extension: ext,
        size: stats.size,
        mtime: stats.mtimeMs,
        isDirectory: false,
        language: this.detectLanguage(ext),
      };
      
      this.files.set(relativePath, entry);
    } catch (error) {
      logger.warn('Failed to index file', { path: relativePath, error });
    }
  }

  private shouldIgnore(relativePath: string): boolean {
    return this.ignorePatterns.some(pattern => pattern.test(relativePath));
  }

  private detectLanguage(extension: string): string | undefined {
    const languageMap: Record<string, string> = {
      '.ts': 'typescript',
      '.tsx': 'typescript',
      '.js': 'javascript',
      '.jsx': 'javascript',
      '.py': 'python',
      '.java': 'java',
      '.go': 'go',
      '.rs': 'rust',
      '.c': 'c',
      '.cpp': 'cpp',
      '.h': 'c',
      '.hpp': 'cpp',
      '.cs': 'csharp',
      '.rb': 'ruby',
      '.php': 'php',
      '.swift': 'swift',
      '.kt': 'kotlin',
      '.scala': 'scala',
      '.vue': 'vue',
      '.svelte': 'svelte',
      '.json': 'json',
      '.yaml': 'yaml',
      '.yml': 'yaml',
      '.md': 'markdown',
      '.html': 'html',
      '.css': 'css',
      '.scss': 'scss',
      '.less': 'less',
      '.sql': 'sql',
      '.sh': 'shell',
      '.bash': 'shell',
    };
    
    return languageMap[extension];
  }

  private generateVersion(): string {
    return `v${Date.now()}`;
  }

  getFile(path: string): FileEntry | undefined {
    return this.files.get(path);
  }

  getAllFiles(): FileEntry[] {
    return Array.from(this.files.values());
  }

  getFilesByExtension(extension: string): FileEntry[] {
    return Array.from(this.files.values()).filter(
      f => f.extension === extension
    );
  }

  getFilesByLanguage(language: string): FileEntry[] {
    return Array.from(this.files.values()).filter(
      f => f.language === language
    );
  }

  getMetadata(): IndexMetadata | null {
    return this.metadata;
  }

  clear(): void {
    this.files.clear();
    this.metadata = null;
  }
}
```

### 任务3: 文本搜索 (1天)

```typescript
// src/search/text-search.ts

import fs from 'fs/promises';
import { createLogger } from '@ai-digital-life/logger';
import type { FileTreeIndexer } from '../indexer/file-tree';
import type { SearchResult, MatchInfo, SearchOptions } from '../types';

const logger = createLogger('text-search');

export class TextSearch {
  private fileIndexer: FileTreeIndexer;

  constructor(fileIndexer: FileTreeIndexer) {
    this.fileIndexer = fileIndexer;
  }

  async search(options: SearchOptions): Promise<SearchResult[]> {
    const { query, filePattern, maxResults = 100, includeContext = true } = options;
    
    const results: SearchResult[] = [];
    const regex = this.buildRegex(query);
    
    const files = this.fileIndexer.getAllFiles();
    
    for (const file of files) {
      if (file.isDirectory) continue;
      if (filePattern && !this.matchPattern(file.path, filePattern)) continue;
      
      try {
        const matches = await this.searchFile(file.path, regex, includeContext);
        
        if (matches.length > 0) {
          results.push({
            type: 'text',
            score: this.calculateScore(matches),
            matches,
          });
        }
        
        if (results.length >= maxResults) break;
      } catch (error) {
        logger.warn('Failed to search file', { path: file.path, error });
      }
    }
    
    return results.sort((a, b) => b.score - a.score).slice(0, maxResults);
  }

  private buildRegex(query: string): RegExp {
    const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
    return new RegExp(escaped, 'gi');
  }

  private async searchFile(
    filePath: string,
    regex: RegExp,
    includeContext: boolean
  ): Promise<MatchInfo[]> {
    const content = await fs.readFile(filePath, 'utf-8');
    const lines = content.split('\n');
    const matches: MatchInfo[] = [];
    
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      let match;
      
      regex.lastIndex = 0;
      
      while ((match = regex.exec(line)) !== null) {
        const matchInfo: MatchInfo = {
          filePath,
          line: i + 1,
          column: match.index + 1,
          endColumn: match.index + match[0].length + 1,
        };
        
        if (includeContext) {
          matchInfo.context = this.getContext(lines, i, 2);
          matchInfo.highlightRanges = [{
            start: match.index,
            end: match.index + match[0].length,
          }];
        }
        
        matches.push(matchInfo);
      }
    }
    
    return matches;
  }

  private getContext(lines: string[], index: number, padding: number): string {
    const start = Math.max(0, index - padding);
    const end = Math.min(lines.length - 1, index + padding);
    
    return lines.slice(start, end + 1).join('\n');
  }

  private matchPattern(filePath: string, pattern: string): boolean {
    const regex = new RegExp(
      pattern.replace(/\*/g, '.*').replace(/\?/g, '.')
    );
    return regex.test(filePath);
  }

  private calculateScore(matches: MatchInfo[]): number {
    return matches.length;
  }
}
```

```typescript
import { createLogger } from '@ai-digital-life/logger';
import type { FileTreeIndexer } from './file-tree';

const logger = createLogger('symbol-indexer');

export class SymbolIndexer {
  private fileIndexer: FileTreeIndexer;
  private symbols = new Map<string, SymbolInfo[]>();
  private imports = new Map<string, ImportInfo[]>();
  private exports = new Map<string, ExportInfo[]>();
  private dependencyGraph: DependencyGraph | null = null;

  constructor(fileIndexer: FileTreeIndexer) {
    this.fileIndexer = fileIndexer;
  }

  async index(): Promise<void> {
    logger.info('Starting symbol indexing');
    
    this.symbols.clear();
    this.imports.clear();
    this.exports.clear();
    
    const files = this.fileIndexer.getAllFiles();
    
    for (const file of files) {
      if (!file.language) continue;
      
      try {
        await this.indexFile(file.path, file.language);
      } catch (error) {
        logger.warn('Failed to index symbols', { path: file.path, error });
      }
    }
    
    this.buildDependencyGraph();
    
    logger.info('Symbol indexing complete', {
      symbolCount: this.getTotalSymbolCount(),
    });
  }

  private async indexFile(filePath: string, language: string): Promise<void> {
    switch (language) {
      case 'typescript':
      case 'javascript':
        await this.indexTypeScript(filePath);
        break;
      case 'python':
        await this.indexPython(filePath);
        break;
      default:
        // 通用索引
        break;
    }
  }

  private async indexTypeScript(filePath: string): Promise<void> {
    // 实际实现需要使用 TypeScript Compiler API
    // 这里提供简化版本
    const fs = await import('fs/promises');
    const content = await fs.readFile(filePath, 'utf-8');
    
    const symbols: SymbolInfo[] = [];
    const imports: ImportInfo[] = [];
    const exports: ExportInfo[] = [];
    
    // 简单的正则匹配（实际应使用 AST）
    const functionRegex = /(?:export\s+)?(?:async\s+)?function\s+(\w+)/g;
    const classRegex = /(?:export\s+)?class\s+(\w+)/g;
    const interfaceRegex = /(?:export\s+)?interface\s+(\w+)/g;
    const constRegex = /(?:export\s+)?const\s+(\w+)/g;
    
    let match;
    
    while ((match = functionRegex.exec(content)) !== null) {
      symbols.push({
        name: match[1],
        kind: 11, // Function
        filePath,
        line: this.getLineNumber(content, match.index),
        column: match.index - this.getLineStart(content, match.index),
      });
    }
    
    while ((match = classRegex.exec(content)) !== null) {
      symbols.push({
        name: match[1],
        kind: 4, // Class
        filePath,
        line: this.getLineNumber(content, match.index),
        column: match.index - this.getLineStart(content, match.index),
      });
    }
    
    while ((match = interfaceRegex.exec(content)) !== null) {
      symbols.push({
        name: match[1],
        kind: 10, // Interface
        filePath,
        line: this.getLineNumber(content, match.index),
        column: match.index - this.getLineStart(content, match.index),
      });
    }
    
    this.symbols.set(filePath, symbols);
    this.imports.set(filePath, imports);
    this.exports.set(filePath, exports);
  }

  private async indexPython(filePath: string): Promise<void> {
    // Python 符号索引实现
  }

  private getLineNumber(content: string, index: number): number {
    return content.substring(0, index).split('\n').length;
  }

  private getLineStart(content: string, index: number): number {
    const lines = content.substring(0, index).split('\n');
    return index - lines[lines.length - 1].length;
  }

  private buildDependencyGraph(): void {
    const files = new Map<string, Set<string>>();
    const reverse = new Map<string, Set<string>>();
    
    for (const [filePath, imports] of this.imports) {
      const deps = new Set<string>();

          if (!reverse.has(resolvedPath)) {
            reverse.set(resolvedPath, new Set());
          }
          reverse.get(resolvedPath)!.add(filePath);
        }
      }
      
      files.set(filePath, deps);

  }

    return count;
  }

    return this.symbols.get(filePath) ?? [];
```

### 任务5: 仓库地图 (2天)

```typescript
// src/map/repo-map.ts

import { createLogger } from '@ai-digital-life/logger';
import type { FileTreeIndexer } from '../indexer/file-tree';
import type { SymbolIndexer } from '../indexer/symbol-index';
import type { RepoMap, DirectoryInfo, ModuleInfo } from '../types';

const logger = createLogger('repo-map');

export class RepoMapGenerator {
  private fileIndexer: FileTreeIndexer;
  private symbolIndexer: SymbolIndexer;

  constructor(fileIndexer: FileTreeIndexer, symbolIndexer: SymbolIndexer) {
    this.fileIndexer = fileIndexer;
    this.symbolIndexer = symbolIndexer;
  }

  async generate(rootPath: string): Promise<RepoMap> {
    logger.info('Generating repository map');
    
    const files = this.fileIndexer.getAllFiles();
    
    const directories = this.analyzeDirectories(files);
    const entryPoints = this.findEntryPoints(files);
    const mainModules = this.findMainModules(files);
    const tests = this.findTests(files);
    const configs = this.findConfigs(files);
    const summary = await this.generateSummary(rootPath, {
      directories,
      entryPoints,
      mainModules,
      tests,
      configs,
    });
    
    return {
      summary,
      directories,
      entryPoints,
      mainModules,
      tests,
      configs,
    };
  }

  private analyzeDirectories(files: ReturnType<FileTreeIndexer['getAllFiles']>): DirectoryInfo[] {
    const dirMap = new Map<string, { files: string[]; count: number }>();
    
    for (const file of files) {
      const dir = this.getDirectory(file.path);
      
      if (!dirMap.has(dir)) {
        dirMap.set(dir, { files: [], count: 0 });
      }
      
      const entry = dirMap.get(dir)!;
      entry.files.push(file.path);
      entry.count++;
    }
    
    return Array.from(dirMap.entries())
      .filter(([_, data]) => data.count >= 3)
      .map(([path, data]) => ({
        path,
        name: path.split('/').pop() ?? path,
        fileCount: data.count,
        mainFiles: data.files.slice(0, 5),
      }))
      .sort((a, b) => b.fileCount - a.fileCount)
      .slice(0, 20);
  }

  private findEntryPoints(files: ReturnType<FileTreeIndexer['getAllFiles']>): string[] {
    const entryPatterns = [
      'index.ts', 'index.js', 'index.tsx', 'index.jsx',
      'main.ts', 'main.js', 'main.tsx', 'main.jsx',
      'app.ts', 'app.js', 'app.tsx', 'app.jsx',
      'server.ts', 'server.js',
      '__init__.py', 'main.py', 'app.py',
    ];
    
    return files
      .filter(f => entryPatterns.includes(f.name))
      .map(f => f.path);
  }

  private findMainModules(files: ReturnType<FileTreeIndexer['getAllFiles']>): ModuleInfo[] {
    const modules: ModuleInfo[] = [];
    
    for (const file of files) {
      const symbols = this.symbolIndexer.getSymbols(file.path);
      const exports = symbols.filter(s => 
        s.kind === 4 || s.kind === 10 || s.kind === 11 // Class, Interface, Function
      );
      
      if (exports.length > 0) {
        modules.push({
          name: file.name.replace(/\.[^.]+$/, ''),
          path: file.path,
          exports: exports.slice(0, 10).map(s => s.name),
          imports: [],
        });
      }
    }
    
    return modules.slice(0, 50);
  }

  private findTests(files: ReturnType<FileTreeIndexer['getAllFiles']>): string[] {
    return files
      .filter(f => 
        f.path.includes('test') || 
        f.path.includes('spec') ||
        f.path.includes('__tests__') ||
        f.name.endsWith('.test.ts') ||
        f.name.endsWith('.test.js') ||
        f.name.endsWith('.spec.ts') ||
        f.name.endsWith('.spec.js')
      )
      .map(f => f.path)
      .slice(0, 20);
  }

  private findConfigs(files: ReturnType<FileTreeIndexer['getAllFiles']>): string[] {
    const configPatterns = [
      'package.json', 'tsconfig.json', 'jsconfig.json',
      '.eslintrc', '.prettierrc', '.babelrc',
      'vite.config', 'webpack.config', 'rollup.config',
      'pyproject.toml', 'setup.py', 'requirements.txt',
      'Cargo.toml', 'go.mod',
    ];
    
    return files
      .filter(f => configPatterns.some(p => f.name.includes(p)))
      .map(f => f.path);
  }

  private getDirectory(filePath: string): string {
    const parts = filePath.split('/');
    return parts.length > 1 ? parts.slice(0, -1).join('/') : '';
  }

  private async generateSummary(
    rootPath: string,
    map: Partial<RepoMap>
  ): Promise<string> {
    const lines: string[] = [];
    
    lines.push(`# Repository Overview`);
    lines.push(``);
    lines.push(`Total files: ${this.fileIndexer.getAllFiles().length}`);
    lines.push(``);
    
    if (map.entryPoints && map.entryPoints.length > 0) {
      lines.push(`## Entry Points`);
      for (const entry of map.entryPoints.slice(0, 5)) {
        lines.push(`- ${entry}`);
      }
      lines.push(``);
    }
    
    if (map.directories && map.directories.length > 0) {
      lines.push(`## Main Directories`);
      for (const dir of map.directories.slice(0, 10)) {
        lines.push(`- ${dir.path} (${dir.fileCount} files)`);
      }
      lines.push(``);
    }
    
    return lines.join('\n');
  }
```

### 任务6: 相关性排序 (1天)

```typescript
// src/ranking/relevance.ts

import type { FileEntry } from '../types';

export interface RankingFactors {
  queryMatch: number;
  dependencyDistance: number;
  recentModification: number;
  isConfig: number;
  isTest: number;
  symbolCount: number;
}

const DEFAULT_FACTORS: RankingFactors = {
  queryMatch: 10,
  dependencyDistance: 5,
  recentModification: 3,
  isConfig: 2,
  isTest: 1,
  symbolCount: 2,
};

export class RelevanceRanker {
  private factors: RankingFactors;

  constructor(factors: Partial<RankingFactors> = {}) {
    this.factors = { ...DEFAULT_FACTORS, ...factors };
  }

  rankFiles(
    files: FileEntry[],
    options: {
      query?: string;
      currentFile?: string;
      dependencyGraph?: Map<string, Set<string>>;
    }
  ): FileEntry[] {
    const scored = files.map(file => ({
      file,
      score: this.calculateScore(file, options),
    }));
    
    return scored
      .sort((a, b) => b.score - a.score)
      .map(s => s.file);
  }

  private calculateScore(
    file: FileEntry,
    options: {
      query?: string;
      currentFile?: string;
      dependencyGraph?: Map<string, Set<string>>;
    }
  ): number {
    let score = 0;
    
    // 查询匹配
    if (options.query) {
      score += this.queryMatchScore(file, options.query) * this.factors.queryMatch;
    }
    
    // 依赖距离
    if (options.currentFile && options.dependencyGraph) {
      score += this.dependencyScore(file.path, options.currentFile, options.dependencyGraph) 
        * this.factors.dependencyDistance;
    }
    
    // 最近修改
    score += this.recencyScore(file.mtime) * this.factors.recentModification;
    
    // 配置文件
    if (this.isConfigFile(file)) {
      score += this.factors.isConfig;
    }
    
    // 测试文件
    if (this.isTestFile(file)) {
      score += this.factors.isTest;
    }
    
    return score;
  }

  private queryMatchScore(file: FileEntry, query: string): number {
    const lowerQuery = query.toLowerCase();
    const lowerName = file.name.toLowerCase();
    const lowerPath = file.path.toLowerCase();
    
    if (lowerName === lowerQuery) return 1;
    if (lowerName.startsWith(lowerQuery)) return 0.8;
    if (lowerName.includes(lowerQuery)) return 0.6;
    if (lowerPath.includes(lowerQuery)) return 0.4;
    
    return 0;
  }

  private dependencyScore(
    filePath: string,
    currentFile: string,
    graph: Map<string, Set<string>>
  ): number {
    if (filePath === currentFile) return 1;
    
    const deps = graph.get(currentFile);
    if (deps?.has(filePath)) return 0.8;
    
    const reverseDeps = graph.get(filePath);
    if (reverseDeps?.has(currentFile)) return 0.6;
    
    return 0;
  }

  private recencyScore(mtime: number): number {
    const age = Date.now() - mtime;
    const days = age / (1000 * 60 * 60 * 24);
    
    if (days < 1) return 1;
    if (days < 7) return 0.7;
    if (days < 30) return 0.4;
    
    return 0.1;
  }

  private isConfigFile(file: FileEntry): boolean {
    const configPatterns = [
      'config', 'settings', '.json', '.yaml', '.yml', '.toml'
    ];
    return configPatterns.some(p => file.name.toLowerCase().includes(p));
  }

  private isTestFile(file: FileEntry): boolean {
    return file.path.includes('test') || file.path.includes('spec');
  }
}
```
