# 持久化与缓存策略实施计划

## 模块概述

**包名**: `@ai-digital-life/storage`

**优先级**: P0 - 数据安全与性能的基础保障

**目标**: 定义哪些数据必须可靠落盘，哪些数据可以短期缓存，以及它们如何失效、恢复和清理。

## 设计原则

- 持久化数据必须服务恢复、审计和稳定性
- 缓存数据必须服务性能，而不能成为唯一真实来源
- 敏感数据与普通缓存必须分层处理

```typescript
import Database from 'better-sqlite3';
import path from 'path';
import fs from 'fs/promises';

export interface DatabaseConfig {
  dataDir: string;
  filename: string;
}

export class DatabaseConnection {
  private db: Database.Database | null = null;
  private config: DatabaseConfig;

  constructor(config: DatabaseConfig) {
    this.config = config;
  }

  async initialize(): Promise<void> {
    await fs.mkdir(this.config.dataDir, { recursive: true });
    
    const dbPath = path.join(this.config.dataDir, this.config.filename);
    this.db = new Database(dbPath);
    
    this.db.pragma('journal_mode = WAL');
    this.db.pragma('synchronous = NORMAL');
    
    await this.runMigrations();
  }

  private async runMigrations(): Promise<void> {
    this.db!.exec(`
      CREATE TABLE IF NOT EXISTS migrations (
        id INTEGER PRIMARY KEY,
        name TEXT NOT NULL,
        applied_at INTEGER NOT NULL
      );
      
      CREATE TABLE IF NOT EXISTS workspaces (
        id TEXT PRIMARY KEY,
        path TEXT NOT NULL,
        name TEXT NOT NULL,
        config TEXT,
        created_at INTEGER NOT NULL,
        updated_at INTEGER NOT NULL
      );
      
      CREATE TABLE IF NOT EXISTS sessions (
        id TEXT PRIMARY KEY,
        workspace_id TEXT NOT NULL,
        summary TEXT,
        checkpoint_id TEXT,
        created_at INTEGER NOT NULL,
        updated_at INTEGER NOT NULL,
        FOREIGN KEY (workspace_id) REFERENCES workspaces(id)
      );
      
      CREATE TABLE IF NOT EXISTS checkpoints (
        id TEXT PRIMARY KEY,
        session_id TEXT NOT NULL,
        type TEXT NOT NULL,
        data TEXT,
        created_at INTEGER NOT NULL,
        FOREIGN KEY (session_id) REFERENCES sessions(id)
      );
      
      CREATE TABLE IF NOT EXISTS approvals (
        id TEXT PRIMARY KEY,
        task_id TEXT NOT NULL,
        risk_level TEXT NOT NULL,
        decision TEXT,
        reason TEXT,
        created_at INTEGER NOT NULL,
        decided_at INTEGER,
        FOREIGN KEY (task_id) REFERENCES tasks(id)
      );
      
      CREATE TABLE IF NOT EXISTS tasks (
        id TEXT PRIMARY KEY,
        workspace_id TEXT,
        session_id TEXT,
        type TEXT NOT NULL,
        status TEXT NOT NULL,
        description TEXT,
        result TEXT,
        error TEXT,
        created_at INTEGER NOT NULL,
        updated_at INTEGER NOT NULL,
        completed_at INTEGER
      );
      
      CREATE INDEX IF NOT EXISTS idx_sessions_workspace ON sessions(workspace_id);
      CREATE INDEX IF NOT EXISTS idx_checkpoints_session ON checkpoints(session_id);
      CREATE INDEX IF NOT EXISTS idx_tasks_workspace ON tasks(workspace_id);
      CREATE INDEX IF NOT EXISTS idx_tasks_status ON tasks(status);
    `);
  }

  get<T = unknown>(sql: string, params: unknown[] = []): T | undefined {
    const stmt = this.db!.prepare(sql);
    return stmt.get(...params) as T | undefined;
  }

  all<T = unknown>(sql: string, params: unknown[] = []): T[] {
    const stmt = this.db!.prepare(sql);
    return stmt.all(...params) as T[];
  }

  run(sql: string, params: unknown[] = []): Database.RunResult {
    const stmt = this.db!.prepare(sql);
    return stmt.run(...params);
  }

  transaction<T>(fn: () => T): T {
    return this.db!.transaction(fn)();
  }

  close(): void {
    this.db?.close();
    this.db = null;
  }
}
```

### 任务3: 缓存系统 (2天)

```typescript
// src/cache/memory-cache.ts

export interface CacheEntry<T> {
  value: T;
  createdAt: number;
  ttl: number;
  version?: string;
}

export class MemoryCache {
  private cache = new Map<string, CacheEntry<unknown>>();
  private maxSize: number;
  private currentSize = 0;

  constructor(maxSize = 1000) {
    this.maxSize = maxSize;
  }

  get<T>(key: string): T | undefined {
    const entry = this.cache.get(key);
    
    if (!entry) return undefined;
    
    if (this.isExpired(entry)) {
      this.delete(key);
      return undefined;
    }
    
    return entry.value as T;
  }

  set<T>(key: string, value: T, ttl: number, version?: string): void {
    if (this.cache.has(key)) {
      this.delete(key);
    }
    
    while (this.currentSize >= this.maxSize) {
      this.evictOldest();
    }
    
    this.cache.set(key, {
      value,
      createdAt: Date.now(),
      ttl,
      version,
    });
    
    this.currentSize++;
  }

  delete(key: string): boolean {
    const existed = this.cache.delete(key);
    if (existed) {
      this.currentSize--;
    }
    return existed;
  }

  has(key: string): boolean {
    const entry = this.cache.get(key);
    if (!entry) return false;
    
    if (this.isExpired(entry)) {
      this.delete(key);
      return false;
    }
    
    return true;
  }

  clear(): void {
    this.cache.clear();
    this.currentSize = 0;
  }

  invalidateByVersion(version: string): number {
    let count = 0;
    
    for (const [key, entry] of this.cache) {
      if (entry.version && entry.version !== version) {
        this.delete(key);
        count++;
      }
    }
    
    return count;
  }

  private isExpired(entry: CacheEntry<unknown>): boolean {
    return Date.now() - entry.createdAt > entry.ttl;
  }

  private evictOldest(): void {
    let oldestKey: string | null = null;
    let oldestTime = Infinity;
    
    for (const [key, entry] of this.cache) {
      if (entry.createdAt < oldestTime) {
        oldestTime = entry.createdAt;
        oldestKey = key;
      }
    }
    
    if (oldestKey) {
      this.delete(oldestKey);
    }
  }
}
```

```typescript
// src/cache/file-cache.ts

import fs from 'fs/promises';
import path from 'path';
import { createHash } from 'crypto';

export interface FileCacheOptions {
  cacheDir: string;
  maxAge: number;
  maxSize: number;
}

export class FileCache {
  private options: FileCacheOptions;

  constructor(options: FileCacheOptions) {
    this.options = options;
  }

  async initialize(): Promise<void> {
    await fs.mkdir(this.options.cacheDir, { recursive: true });
  }

  async get<T>(key: string): Promise<T | undefined> {
    const filePath = this.getFilePath(key);
    
    try {
      const content = await fs.readFile(filePath, 'utf-8');
      const entry = JSON.parse(content);
      
      if (Date.now() > entry.expiresAt) {
        await this.delete(key);
        return undefined;
      }
      
      return entry.value as T;
    } catch {
      return undefined;
    }
  }

  async set<T>(key: string, value: T, ttl?: number): Promise<void> {
    const filePath = this.getFilePath(key);
    const entry = {
      value,
      createdAt: Date.now(),
      expiresAt: Date.now() + (ttl ?? this.options.maxAge),
    };
    
    await fs.writeFile(filePath, JSON.stringify(entry), 'utf-8');
  }

  async delete(key: string): Promise<boolean> {
    const filePath = this.getFilePath(key);
    
    try {
      await fs.unlink(filePath);
      return true;
    } catch {
      return false;
    }
  }

  async clear(): Promise<void> {
    const files = await fs.readdir(this.options.cacheDir);
    
    await Promise.all(
      files.map(file => 
        fs.unlink(path.join(this.options.cacheDir, file)).catch(() => {})
      )
    );
  }

  async cleanup(): Promise<number> {
    const files = await fs.readdir(this.options.cacheDir);
    let cleaned = 0;
    
    for (const file of files) {
      const filePath = path.join(this.options.cacheDir, file);
      
      try {
        const content = await fs.readFile(filePath, 'utf-8');
        const entry = JSON.parse(content);
        
        if (Date.now() > entry.expiresAt) {
          await fs.unlink(filePath);
          cleaned++;
        }
      } catch {
        await fs.unlink(filePath).catch(() => {});
        cleaned++;
      }
    }
    
    return cleaned;
  }

  private getFilePath(key: string): string {
    const hash = createHash('sha256').update(key).digest('hex');
    return path.join(this.options.cacheDir, `${hash}.cache`);
  }
}
```

```typescript
// src/snapshot/checkpoint.ts

import { generateId } from '@ai-digital-life/kernel';
import type { DatabaseConnection } from '../database/connection';

export type CheckpointType = 
  | 'pre-task'       // 任务开始前
  | 'pre-patch'      // 应用patch前
  | 'pre-command'    // 执行高风险命令前
  | 'pre-batch'      // 批量文件改动前
  | 'manual';        // 手动创建

export interface Checkpoint {
  id: string;
  sessionId: string;
  type: CheckpointType;
  description?: string;
  data: {
    files: Record<string, string>;
    gitStatus?: {
      branch: string;
      staged: string[];
      unstaged: string[];
      untracked: string[];
    };
    metadata: Record<string, unknown>;
  };
  createdAt: number;
}

export class CheckpointManager {
  private db: DatabaseConnection;
  private snapshotDir: string;

  constructor(db: DatabaseConnection, snapshotDir: string) {
    this.db = db;
    this.snapshotDir = snapshotDir;
  }

  async create(
    sessionId: string,
    type: CheckpointType,
    description?: string
  ): Promise<Checkpoint> {
    const id = generateId();
    const checkpoint: Checkpoint = {
      id,
      sessionId,
      type,
      description,
      data: {
        files: {},
        metadata: {},
      },
      createdAt: Date.now(),
    };

    this.db.run(
      `INSERT INTO checkpoints (id, session_id, type, data, created_at)
       VALUES (?, ?, ?, ?, ?)`,
      [id, sessionId, type, JSON.stringify(checkpoint.data), checkpoint.createdAt]
    );

    return checkpoint;
  }

  async load(id: string): Promise<Checkpoint | undefined> {
    const row = this.db.get<{ id: string; session_id: string; type: string; data: string; created_at: number }>(
      'SELECT * FROM checkpoints WHERE id = ?',
      [id]
    );

    if (!row) return undefined;

    return {
      id: row.id,
      sessionId: row.session_id,
      type: row.type as CheckpointType,
      data: JSON.parse(row.data),
      createdAt: row.created_at,
    };
  }

  async getLatest(sessionId: string): Promise<Checkpoint | undefined> {
    const row = this.db.get<{ id: string; session_id: string; type: string; data: string; created_at: number }>(
      'SELECT * FROM checkpoints WHERE session_id = ? ORDER BY created_at DESC LIMIT 1',
      [sessionId]
    );

    if (!row) return undefined;

    return {
      id: row.id,
      sessionId: row.session_id,
      type: row.type as CheckpointType,
      data: JSON.parse(row.data),
      createdAt: row.created_at,
    };
  }

  async list(sessionId: string, limit = 10): Promise<Checkpoint[]> {
    const rows = this.db.all<{ id: string; session_id: string; type: string; data: string; created_at: number }>(
      'SELECT * FROM checkpoints WHERE session_id = ? ORDER BY created_at DESC LIMIT ?',
      [sessionId, limit]
    );

    return rows.map(row => ({
      id: row.id,
      sessionId: row.session_id,
      type: row.type as CheckpointType,
      data: JSON.parse(row.data),
      createdAt: row.created_at,
    }));
  }

  async delete(id: string): Promise<boolean> {
    const result = this.db.run('DELETE FROM checkpoints WHERE id = ?', [id]);
    return result.changes > 0;
  }

  async deleteOlderThan(sessionId: string, timestamp: number): Promise<number> {
    const result = this.db.run(
      'DELETE FROM checkpoints WHERE session_id = ? AND created_at < ?',
      [sessionId, timestamp]
    );
    return result.changes;
  }
}
```

```typescript
// src/security/keychain.ts

import { safeStorage } from 'electron';

export interface KeychainProvider {
  store(key: string, value: string): Promise<void>;
  retrieve(key: string): Promise<string | null>;
  delete(key: string): Promise<boolean>;
}

export class SystemKeychain implements KeychainProvider {
  private serviceName: string;

  constructor(serviceName: string) {
    this.serviceName = serviceName;
  }

  async store(key: string, value: string): Promise<void> {
    const encrypted = safeStorage.encryptString(value);
    localStorage.setItem(
      `${this.serviceName}:${key}`,
      encrypted.toString('base64')
    );
  }

  async retrieve(key: string): Promise<string | null> {
    const encrypted = localStorage.getItem(`${this.serviceName}:${key}`);
    
    if (!encrypted) return null;
    
    try {
      const buffer = Buffer.from(encrypted, 'base64');
      return safeStorage.decryptString(buffer);
    } catch {
      return null;
    }
  }

  async delete(key: string): Promise<boolean> {
    const fullKey = `${this.serviceName}:${key}`;
    
    if (localStorage.getItem(fullKey) === null) {
      return false;
    }
    
    localStorage.removeItem(fullKey);
    return true;
  }
}
```

```typescript
// src/security/sanitizer.ts

const SENSITIVE_PATTERNS = [
  /api[_-]?key/i,
  /password/i,
  /secret/i,
  /token/i,
  /credential/i,
  /auth[_-]?token/i,
  /access[_-]?token/i,
  /refresh[_-]?token/i,
  /private[_-]?key/i,
];

const PATH_PATTERNS = [
  /\/Users\/[^/]+\//g,
  /\/home\/[^/]+\//g,
  /C:\\Users\\[^\\]+\\/g,
  /\\Users\\[^\\]+\\/g,
];

export function sanitizeForLog(data: unknown): unknown {
  if (typeof data === 'string') {
    return sanitizeString(data);
  }
  
  if (Array.isArray(data)) {
    return data.map(sanitizeForLog);
  }
  
  if (data && typeof data === 'object') {
    const sanitized: Record<string, unknown> = {};
    
    for (const [key, value] of Object.entries(data)) {
      if (isSensitiveKey(key)) {
        sanitized[key] = '[REDACTED]';
      } else {
        sanitized[key] = sanitizeForLog(value);
      }
    }
    
    return sanitized;
  }
  
  return data;
}

function sanitizeString(str: string): string {
  let result = str;
  
  for (const pattern of PATH_PATTERNS) {
    result = result.replace(pattern, '[USER_PATH]/');
  }
  
  return result;
}

function isSensitiveKey(key: string): boolean {
  return SENSITIVE_PATTERNS.some(pattern => pattern.test(key));
}
```

```typescript
import { describe, it, expect, beforeEach } from 'vitest';
import { MemoryCache } from '../src/cache/memory-cache';

describe('MemoryCache', () => {
```

## 验收标准

- [ ] 重启后能恢复必要工作区与会话状态
- [ ] 缓存失效不会导致错误结果长期存在
- [ ] 快照可用于回滚最近一次高风险操作
- [ ] 敏感数据不会以明文形式落盘
- [ ] 数据库支持事务操作
- [ ] 缓存支持TTL过期和版本失效
