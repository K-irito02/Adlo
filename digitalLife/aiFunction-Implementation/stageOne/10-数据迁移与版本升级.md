# 数据迁移与版本升级实施计划

## 模块概述

**包名**: `@ai-digital-life/migration`

**优先级**: P1 - 版本迭代的必备能力，应在第一个正式版本发布前完成

**目标**: 确保应用在版本迭代过程中，用户数据、配置和状态能够平滑迁移，避免因版本升级导致的数据丢失或功能异常。

## 设计原则

- 版本检测自动进行
- 迁移脚本幂等可重试
- 失败自动回滚
- 数据完整性可验证

### 任务2: 版本检测与比较 (1天)

```typescript
// src/version/detector.ts

import fs from 'fs/promises';
import path from 'path';
import { createLogger } from '@ai-digital-life/logger';
import type { Version, VersionInfo, VersionChange } from '../types';

const logger = createLogger('version-detector');

export class VersionDetector {
  private dataDir: string;
  private currentVersion: Version;

  constructor(dataDir: string, currentVersion: Version) {
    this.dataDir = dataDir;
    this.currentVersion = currentVersion;
  }

  async detect(): Promise<VersionInfo> {
    const storedVersion = await this.getStoredVersion();
    
    if (!storedVersion) {
      await this.saveVersion(this.currentVersion);
      return {
        current: this.currentVersion,
        updateAvailable: false,
      };
    }

    const change = this.compareVersions(storedVersion, this.currentVersion);
    
    if (change !== 'none') {
      logger.info('Version change detected', {
        from: this.versionToString(storedVersion),
        to: this.versionToString(this.currentVersion),
        change,
      });
    }

    return {
      current: this.currentVersion,
      change,
      updateAvailable: false,
    };
  }

  async getStoredVersion(): Promise<Version | null> {
    try {
      const versionPath = path.join(this.dataDir, '.version');
      const content = await fs.readFile(versionPath, 'utf-8');
      return this.parseVersion(content.trim());
    } catch {
      return null;
    }
  }

  async saveVersion(version: Version): Promise<void> {
    const versionPath = path.join(this.dataDir, '.version');
    await fs.mkdir(this.dataDir, { recursive: true });
    await fs.writeFile(versionPath, this.versionToString(version), 'utf-8');
  }

  parseVersion(versionString: string): Version {
    const match = versionString.match(
      /^v?(\d+)\.(\d+)\.(\d+)(?:-([a-zA-Z0-9.-]+))?(?:\+([a-zA-Z0-9.-]+))?$/
    );

    if (!match) {
      throw new Error(`Invalid version string: ${versionString}`);
    }

    return {
      major: parseInt(match[1], 10),
      minor: parseInt(match[2], 10),
      patch: parseInt(match[3], 10),
      prerelease: match[4],
      build: match[5],
    };
  }

  versionToString(version: Version): string {
    let str = `${version.major}.${version.minor}.${version.patch}`;
    if (version.prerelease) {
      str += `-${version.prerelease}`;
    }
    if (version.build) {
      str += `+${version.build}`;
    }
    return str;
  }

  compareVersions(a: Version, b: Version): VersionChange {
    if (a.major !== b.major) return 'major';
    if (a.minor !== b.minor) return 'minor';
    if (a.patch !== b.patch) return 'patch';
    if (a.prerelease !== b.prerelease) return 'prerelease';
    return 'none';
  }

  isUpgrade(from: Version, to: Version): boolean {
    const fromStr = this.versionToString(from);
    const toStr = this.versionToString(to);
    return fromStr < toStr;
  }
}
```

### 任务3: 备份管理 (1天)

```typescript
// src/migration/backup.ts

import fs from 'fs/promises';
import path from 'path';
import { createLogger } from '@ai-digital-life/logger';
import type { BackupInfo } from '../types';

const logger = createLogger('backup-manager');

export class BackupManager {
  private backupDir: string;
  private dataDir: string;

  constructor(backupDir: string, dataDir: string) {
    this.backupDir = backupDir;
    this.dataDir = dataDir;
  }

  async createFullBackup(version: string): Promise<BackupInfo> {
    const id = `backup_${Date.now()}`;
    const backupPath = path.join(this.backupDir, id);

    logger.info('Creating full backup', { id, version });

    await fs.mkdir(backupPath, { recursive: true });

    const components = ['config', 'database', 'cache'];
    
    for (const component of components) {
      const sourcePath = path.join(this.dataDir, component);
      const targetPath = path.join(backupPath, component);
      
      try {
        await this.copyDirectory(sourcePath, targetPath);
      } catch (error) {
        logger.warn('Failed to backup component', { component, error });
      }
    }

    const size = await this.getDirectorySize(backupPath);

    const info: BackupInfo = {
      id,
      version,
      timestamp: Date.now(),
      path: backupPath,
      size,
      type: 'full',
      components,
    };

    await this.saveBackupInfo(info);

    logger.info('Full backup created', { id, size });

    return info;
  }

  async createPartialBackup(
    version: string,
    components: string[]
  ): Promise<BackupInfo> {
    const id = `backup_${Date.now()}`;
    const backupPath = path.join(this.backupDir, id);

    logger.info('Creating partial backup', { id, version, components });

    await fs.mkdir(backupPath, { recursive: true });

    for (const component of components) {
      const sourcePath = path.join(this.dataDir, component);
      const targetPath = path.join(backupPath, component);
      
      try {
        await this.copyDirectory(sourcePath, targetPath);
      } catch (error) {
        logger.warn('Failed to backup component', { component, error });
      }
    }

    const size = await this.getDirectorySize(backupPath);

    const info: BackupInfo = {
      id,
      version,
      timestamp: Date.now(),
      path: backupPath,
      size,
      type: 'partial',
      components,
    };

    await this.saveBackupInfo(info);

    return info;
  }

  async restore(backupId: string): Promise<boolean> {
    const info = await this.getBackupInfo(backupId);
    
    if (!info) {
      throw new Error(`Backup not found: ${backupId}`);
    }

    logger.info('Restoring from backup', { backupId, version: info.version });

    for (const component of info.components) {
      const sourcePath = path.join(info.path, component);
      const targetPath = path.join(this.dataDir, component);
      
      try {
        await fs.rm(targetPath, { recursive: true, force: true });
        await this.copyDirectory(sourcePath, targetPath);
      } catch (error) {
        logger.error('Failed to restore component', { component, error });
        return false;
      }
    }

    logger.info('Backup restored successfully', { backupId });

    return true;
  }

  async listBackups(): Promise<BackupInfo[]> {
    try {
      const entries = await fs.readdir(this.backupDir);
      const backups: BackupInfo[] = [];

      for (const entry of entries) {
        const info = await this.getBackupInfo(entry);
        if (info) {
          backups.push(info);
        }
      }

      return backups.sort((a, b) => b.timestamp - a.timestamp);
    } catch {
      return [];
    }
  }

  async deleteBackup(backupId: string): Promise<boolean> {
    const info = await this.getBackupInfo(backupId);
    
    if (!info) return false;

    await fs.rm(info.path, { recursive: true, force: true });
    
    logger.info('Backup deleted', { backupId });

    return true;
  }

  async cleanupOldBackups(maxAge: number = 30 * 24 * 60 * 60 * 1000): Promise<number> {
    const backups = await this.listBackups();
    const cutoff = Date.now() - maxAge;
    let deleted = 0;

    for (const backup of backups) {
      if (backup.timestamp < cutoff) {
        await this.deleteBackup(backup.id);
        deleted++;
      }
    }

    logger.info('Old backups cleaned up', { deleted });

    return deleted;
  }

  private async copyDirectory(source: string, target: string): Promise<void> {
    await fs.mkdir(target, { recursive: true });
    
    const entries = await fs.readdir(source, { withFileTypes: true });
    
    for (const entry of entries) {
      const sourcePath = path.join(source, entry.name);
      const targetPath = path.join(target, entry.name);
      
      if (entry.isDirectory()) {
        await this.copyDirectory(sourcePath, targetPath);
      } else {
        await fs.copyFile(sourcePath, targetPath);
      }
    }
  }

  private async getDirectorySize(dirPath: string): Promise<number> {
    let size = 0;
    
    try {
      const entries = await fs.readdir(dirPath, { withFileTypes: true });
      
      for (const entry of entries) {
        const fullPath = path.join(dirPath, entry.name);
        
        if (entry.isDirectory()) {
          size += await this.getDirectorySize(fullPath);
        } else {
          const stats = await fs.stat(fullPath);
          size += stats.size;
        }
      }
    } catch {
      // Ignore errors
    }
    
    return size;
  }

  private async saveBackupInfo(info: BackupInfo): Promise<void> {
    const infoPath = path.join(info.path, 'backup.json');
    await fs.writeFile(infoPath, JSON.stringify(info, null, 2), 'utf-8');
  }

  private async getBackupInfo(backupId: string): Promise<BackupInfo | null> {
    try {
      const infoPath = path.join(this.backupDir, backupId, 'backup.json');
      const content = await fs.readFile(infoPath, 'utf-8');
      return JSON.parse(content);
    } catch {
      return null;
    }
  }
}
```

### 任务4: 迁移执行器 (2天)

```typescript
// src/migration/runner.ts

import { createLogger } from '@ai-digital-life/logger';
import type { DatabaseConnection } from '@ai-digital-life/storage';
import { BackupManager } from './backup';
import { MigrationRegistry } from './registry';
import type { MigrationScript, MigrationRecord, MigrationStatus, Version } from '../types';

const logger = createLogger('migration-runner');

export class MigrationRunner {
  private db: DatabaseConnection;
  private registry: MigrationRegistry;
  private backupManager: BackupManager;

  constructor(
    db: DatabaseConnection,
    registry: MigrationRegistry,
    backupManager: BackupManager
  ) {
    this.db = db;
    this.registry = registry;
    this.backupManager = backupManager;
  }

  async runMigrations(
    fromVersion: Version,
    toVersion: Version
  ): Promise<{ success: boolean; migrations: MigrationRecord[] }> {
    const scripts = this.registry.getScripts(fromVersion, toVersion);
    
    if (scripts.length === 0) {
      logger.info('No migrations needed');
      return { success: true, migrations: [] };
    }

    logger.info('Starting migrations', {
      count: scripts.length,
      from: this.versionToString(fromVersion),
      to: this.versionToString(toVersion),
    });

    const backup = await this.backupManager.createFullBackup(
      this.versionToString(fromVersion)
    );

    const records: MigrationRecord[] = [];

    for (const script of scripts) {
      const record = await this.runScript(script, backup.path);
      records.push(record);

      if (record.status === 'failed') {
        await this.rollback(records, backup.id);
        return { success: false, migrations: records };
      }
    }

    logger.info('All migrations completed successfully');

    return { success: true, migrations: records };
  }

  private async runScript(
    script: MigrationScript,
    backupPath: string
  ): Promise<MigrationRecord> {
    const record: MigrationRecord = {
      id: `migration_${Date.now()}_${script.id}`,
      scriptId: script.id,
      fromVersion: this.versionToString(script.fromVersion),
      toVersion: this.versionToString(script.toVersion),
      type: script.type,
      status: 'running',
      startedAt: Date.now(),
      backupPath,
    };

    this.saveRecord(record);

    try {
      // Pre-check
      if (script.preCheck) {
        const canProceed = await script.preCheck();
        if (!canProceed) {
          throw new Error('Pre-check failed');
        }
      }

      // Run migration
      await script.migrate();

      // Post-check
      if (script.postCheck) {
        const success = await script.postCheck();
        if (!success) {
          throw new Error('Post-check failed');
        }
      }

      record.status = 'completed';
      record.completedAt = Date.now();

      logger.info('Migration completed', { scriptId: script.id });
    } catch (error) {
      record.status = 'failed';
      record.error = (error as Error).message;

      logger.error('Migration failed', error as Error, { scriptId: script.id });
    }

    this.saveRecord(record);

    return record;
  }

  private async rollback(
    records: MigrationRecord[],
    backupId: string
  ): Promise<void> {
    logger.warn('Rolling back migrations', { backupId });

    const failedIndex = records.findIndex(r => r.status === 'failed');
    const toRollback = records.slice(0, failedIndex + 1).reverse();

    for (const record of toRollback) {
      const script = this.registry.getScript(record.scriptId);
      
      if (script) {
        try {
          await script.rollback();
          record.status = 'rolled-back';
          this.saveRecord(record);
        } catch (error) {
          logger.error('Rollback failed', error as Error, { scriptId: script.id });
        }
      }
    }

    await this.backupManager.restore(backupId);
  }

  private saveRecord(record: MigrationRecord): void {
    this.db.run(
      `INSERT OR REPLACE INTO migration_records 
       (id, script_id, from_version, to_version, type, status, started_at, completed_at, error, backup_path)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        record.id,
        record.scriptId,
        record.fromVersion,
        record.toVersion,
        record.type,
        record.status,
        record.startedAt,
        record.completedAt ?? null,
        record.error ?? null,
        record.backupPath ?? null,
      ]
    );
  }

  private versionToString(version: Version): string {
    return `${version.major}.${version.minor}.${version.patch}`;
  }
}
```

```typescript
// src/migration/registry.ts

import type { MigrationScript, Version } from '../types';

export class MigrationRegistry {
  private scripts = new Map<string, MigrationScript>();

  register(script: MigrationScript): void {
    const key = this.getKey(script.fromVersion, script.toVersion);
    this.scripts.set(key, script);
  }

  getScript(scriptId: string): MigrationScript | undefined {
    for (const script of this.scripts.values()) {
      if (script.id === scriptId) {
        return script;
      }
    }
    return undefined;
  }

  getScripts(fromVersion: Version, toVersion: Version): MigrationScript[] {
    const scripts: MigrationScript[] = [];
    let current = { ...fromVersion };

    while (this.compareVersions(current, toVersion) < 0) {
      const nextVersion = this.findNextVersion(current);
      
      if (!nextVersion) {
        break;
      }

      const key = this.getKey(current, nextVersion);
      const script = this.scripts.get(key);
      
      if (script) {
        scripts.push(script);
      }
      
      current = nextVersion;
    }

    return scripts;
  }

  private findNextVersion(version: Version): Version | null {
    let next: Version | null = null;
    let minDiff = Infinity;

    for (const script of this.scripts.values()) {
      if (this.compareVersions(script.fromVersion, version) === 0) {
        const diff = this.versionDiff(script.toVersion, version);
        if (diff < minDiff) {
          minDiff = diff;
          next = script.toVersion;
        }
      }
    }

    return next;
  }

  private getKey(from: Version, to: Version): string {
    return `${this.versionToString(from)}->${this.versionToString(to)}`;
  }

  private versionToString(v: Version): string {
    return `${v.major}.${v.minor}.${v.patch}`;
  }

  private compareVersions(a: Version, b: Version): number {
    if (a.major !== b.major) return a.major - b.major;
    if (a.minor !== b.minor) return a.minor - b.minor;
    return a.patch - b.patch;
  }

  private versionDiff(a: Version, b: Version): number {
    return (a.major - b.major) * 10000 + (a.minor - b.minor) * 100 + (a.patch - b.patch);
  }
}
```

```typescript
// src/validation/validator.ts

import { createLogger } from '@ai-digital-life/logger';
import type { DatabaseConnection } from '@ai-digital-life/storage';

const logger = createLogger('data-validator');

export interface ValidationResult {
  valid: boolean;
  errors: string[];
  warnings: string[];
}

    this.db = db;

    const errors: string[] = [];
    const warnings: string[] = [];

    // 验证数据库完整性

    errors.push(...dbResult.errors);
```

- [ ] 次版本升级：向后兼容，旧配置可继续使用
- [ ] 修订版本升级：无数据变更，热更新成功
- [ ] 跨版本升级：跳过中间版本可正确迁移

### 健壮性验收

- [ ] 迁移过程中断后可恢复
- [ ] 磁盘空间不足时优雅失败
- [ ] 迁移脚本错误不影响原有数据
- [ ] 并发迁移请求正确序列化处理

## 依赖关系

| 依赖模块 | 依赖方式 | 说明 |
|----------|----------|------|
| 配置管理系统 | 强依赖 | 读取版本配置、执行配置迁移 |
| 持久化与缓存策略 | 强依赖 | 数据备份与恢复 |
| 日志与监控系统 | 强依赖 | 迁移过程记录 |
| 错误处理与恢复机制 | 中依赖 | 迁移失败处理 |

## 扩展方向（延后实现）

1. **增量迁移**：支持大数据量增量迁移，减少停机时间
2. **迁移预演**：升级前预演迁移过程，提前发现问题
3. **多版本共存**：支持多版本并行运行，灰度发布
4. **云端同步**：迁移数据云端备份，支持跨设备恢复
5. **迁移可视化**：可视化迁移进度和状态
