# 工作区与会话管理实施计划

## 模块概述

**包名**: `@ai-digital-life/workspace`

**优先级**: P0 - 系统运行的基础上下文

**目标**: 管理"当前项目边界"和"当前任务状态"，为对话、索引、命令执行、变更应用和检查点恢复提供统一上下文。

## 设计原则

- AI是否改对项目，首先取决于工作区边界是否准确
- 任务是否能中断后继续，取决于会话状态是否可恢复
- 变更是否能回滚，取决于是否有稳定的检查点与快照机制

### 任务2: 项目检测器 (1天)

```typescript
// src/workspace/detector.ts

import fs from 'fs/promises';
import path from 'path';
import type { ProjectType } from '../types';

interface ProjectMarker {
  file: string;
  type: ProjectType;
}

const PROJECT_MARKERS: ProjectMarker[] = [
  { file: 'package.json', type: 'node' },
  { file: 'pyproject.toml', type: 'python' },
  { file: 'setup.py', type: 'python' },
  { file: 'requirements.txt', type: 'python' },
  { file: 'pom.xml', type: 'java' },
  { file: 'build.gradle', type: 'java' },
  { file: 'go.mod', type: 'go' },
  { file: 'Cargo.toml', type: 'rust' },
];

export class ProjectDetector {
  async detect(projectPath: string): Promise<ProjectType> {
    for (const marker of PROJECT_MARKERS) {
      const markerPath = path.join(projectPath, marker.file);
      
      try {
        await fs.access(markerPath);
        return marker.type;
      } catch {
        continue;
      }
    }
    
    return 'unknown';
  }

  async findProjectRoot(startPath: string): Promise<string | null> {
    let currentPath = startPath;
    
    while (currentPath !== path.dirname(currentPath)) {
      const type = await this.detect(currentPath);
      
      if (type !== 'unknown') {
        return currentPath;
      }
      
      currentPath = path.dirname(currentPath);
    }
    
    return null;
  }

  async getProjectInfo(projectPath: string): Promise<{
    type: ProjectType;
    name: string;
    hasGit: boolean;
  }> {
    const type = await this.detect(projectPath);
    const name = path.basename(projectPath);
    const hasGit = await this.checkGit(projectPath);
    
    return { type, name, hasGit };
  }

  private async checkGit(projectPath: string): Promise<boolean> {
    try {
      const gitPath = path.join(projectPath, '.git');
      await fs.access(gitPath);
      return true;
    } catch {
      return false;
    }
  }
}
```

### 任务3: 工作区管理 (2天)

```typescript
// src/workspace/workspace.ts

import { generateId } from '@ai-digital-life/kernel';
import { EventEmitter } from '@ai-digital-life/events';
import { createLogger } from '@ai-digital-life/logger';
import type { DatabaseConnection } from '@ai-digital-life/storage';
import { ConfigManager } from '@ai-digital-life/config';
import { ProjectDetector } from './detector';
import type { WorkspaceMeta, WorkspaceStatus, GitStatus } from '../types';

const logger = createLogger('workspace');

export class WorkspaceManager {
  private db: DatabaseConnection;
  private config: ConfigManager;
  private detector: ProjectDetector;
  private emitter: EventEmitter;
  
  private currentWorkspace: WorkspaceMeta | null = null;
  private workspaces = new Map<string, WorkspaceMeta>();

  constructor(
    db: DatabaseConnection,
    config: ConfigManager,
    emitter: EventEmitter
  ) {
    this.db = db;
    this.config = config;
    this.detector = new ProjectDetector();
    this.emitter = emitter;
  }

  async open(workspacePath: string): Promise<WorkspaceMeta> {
    const projectRoot = await this.detector.findProjectRoot(workspacePath);
    const targetPath = projectRoot ?? workspacePath;
    
    const existing = this.db.get<{ id: string }>(
      'SELECT id FROM workspaces WHERE path = ?',
      [targetPath]
    );
    
    if (existing) {
      return this.loadExisting(existing.id, targetPath);
    }
    
    return this.createNew(targetPath);
  }

  private async loadExisting(id: string, path: string): Promise<WorkspaceMeta> {
    const row = this.db.get<{
      id: string;
      path: string;
      name: string;
      config: string;
      created_at: number;
      updated_at: number;
    }>('SELECT * FROM workspaces WHERE id = ?', [id]);
    
    if (!row) {
      throw new Error(`Workspace not found: ${id}`);
    }
    
    const projectInfo = await this.detector.getProjectInfo(path);
    const gitStatus = await this.getGitStatus(path);
    
    const workspace: WorkspaceMeta = {
      id: row.id,
      path: row.path,
      name: row.name,
      type: projectInfo.type,
      status: 'ready',
      gitStatus,
      config: JSON.parse(row.config),
      createdAt: row.created_at,
      updatedAt: row.updated_at,
      lastOpenedAt: Date.now(),
    };
    
    this.db.run(
      'UPDATE workspaces SET updated_at = ? WHERE id = ?',
      [Date.now(), id]
    );
    
    this.currentWorkspace = workspace;
    this.workspaces.set(workspace.id, workspace);
    
    await this.emitter.emit({
      type: 'workspace.opened',
      timestamp: Date.now(),
      payload: { workspaceId: workspace.id, path: workspace.path },
    });
    
    logger.info('Workspace opened', { workspaceId: workspace.id, path });
    
    return workspace;
  }

  private async createNew(path: string): Promise<WorkspaceMeta> {
    const id = generateId();
    const projectInfo = await this.detector.getProjectInfo(path);
    const gitStatus = await this.getGitStatus(path);
    
    const config = await this.config.loadWorkspaceConfig(path);
    
    const workspace: WorkspaceMeta = {
      id,
      path,
      name: projectInfo.name,
      type: projectInfo.type,
      status: 'initializing',
      gitStatus,
      config,
      createdAt: Date.now(),
      updatedAt: Date.now(),
      lastOpenedAt: Date.now(),
    };
    
    this.db.run(
      `INSERT INTO workspaces (id, path, name, config, created_at, updated_at)
       VALUES (?, ?, ?, ?, ?, ?)`,
      [id, path, workspace.name, JSON.stringify(config), workspace.createdAt, workspace.updatedAt]
    );
    
    this.currentWorkspace = workspace;
    this.workspaces.set(workspace.id, workspace);
    
    await this.emitter.emit({
      type: 'workspace.opened',
      timestamp: Date.now(),
      payload: { workspaceId: workspace.id, path: workspace.path },
    });
    
    logger.info('Workspace created', { workspaceId: workspace.id, path });
    
    return workspace;
  }

  async close(workspaceId: string): Promise<void> {
    const workspace = this.workspaces.get(workspaceId);
    
    if (!workspace) return;
    
    this.db.run(
      'UPDATE workspaces SET updated_at = ? WHERE id = ?',
      [Date.now(), workspaceId]
    );
    
    this.workspaces.delete(workspaceId);
    
    if (this.currentWorkspace?.id === workspaceId) {
      this.currentWorkspace = null;
    }
    
    await this.emitter.emit({
      type: 'workspace.closed',
      timestamp: Date.now(),
      payload: { workspaceId },
    });
    
    logger.info('Workspace closed', { workspaceId });
  }

  getCurrent(): WorkspaceMeta | null {
    return this.currentWorkspace;
  }

  get(workspaceId: string): WorkspaceMeta | undefined {
    return this.workspaces.get(workspaceId);
  }

  async setStatus(workspaceId: string, status: WorkspaceStatus): Promise<void> {
    const workspace = this.workspaces.get(workspaceId);
    
    if (!workspace) return;
    
    workspace.status = status;
    workspace.updatedAt = Date.now();
    
    await this.emitter.emit({
      type: 'workspace.status-changed',
      timestamp: Date.now(),
      payload: { workspaceId, status },
    });
  }

  async refreshGitStatus(workspaceId: string): Promise<GitStatus | undefined> {
    const workspace = this.workspaces.get(workspaceId);
    
    if (!workspace) return undefined;
    
    const gitStatus = await this.getGitStatus(workspace.path);
    workspace.gitStatus = gitStatus;
    workspace.updatedAt = Date.now();
    
    await this.emitter.emit({
      type: 'workspace.git-changed',
      timestamp: Date.now(),
      payload: { workspaceId, gitStatus },
    });
    
    return gitStatus;
  }

  private async getGitStatus(projectPath: string): Promise<GitStatus | undefined> {
    // 实际实现需要调用 git 命令
    return undefined;
  }

  async getRecent(limit = 10): Promise<WorkspaceMeta[]> {
    const rows = this.db.all<{
      id: string;
      path: string;
      name: string;
      config: string;
      created_at: number;
      updated_at: number;
    }>(
      'SELECT * FROM workspaces ORDER BY updated_at DESC LIMIT ?',
      [limit]
    );
    
    return rows.map(row => ({
      id: row.id,
      path: row.path,
      name: row.name,
      type: 'unknown' as const,
      status: 'closed' as const,
      config: JSON.parse(row.config),
      createdAt: row.created_at,
      updatedAt: row.updated_at,
      lastOpenedAt: row.updated_at,
    }));
  }
}
```

```typescript
  private currentSession: SessionMeta | null = null;

  constructor(db: DatabaseConnection, emitter: EventEmitter) {
    this.db = db;
    this.emitter = emitter;
  }

  async create(workspaceId: string): Promise<SessionMeta> {
    const id = generateId();
    
    const session: SessionMeta = {
      id,
      workspaceId,
      status: 'active',
      approvalMode: 'semi-auto',
      createdAt: Date.now(),
      updatedAt: Date.now(),
    };
    
    this.db.run(
      `INSERT INTO sessions (id, workspace_id, status, approval_mode, created_at, updated_at)
       VALUES (?, ?, ?, ?, ?, ?)`,
      [id, workspaceId, 'active', 'semi-auto', session.createdAt, session.updatedAt]
    );
    
    this.currentSession = session;
    this.sessions.set(session.id, session);
    
    await this.emitter.emit({
      type: 'session.created',
      timestamp: Date.now(),
      payload: { sessionId: id, workspaceId },
    });
    
    logger.info('Session created', { sessionId: id, workspaceId });
    
    return session;
  }

  async restore(sessionId: string): Promise<SessionMeta> {
    const row = this.db.get<{
      id: string;
      workspace_id: string;
      status: string;
      summary: string | null;
      checkpoint_id: string | null;
      approval_mode: string;
      created_at: number;
      updated_at: number;
    }>('SELECT * FROM sessions WHERE id = ?', [sessionId]);
    
    if (!row) {
      throw new Error(`Session not found: ${sessionId}`);
    }
    
    const session: SessionMeta = {
      id: row.id,
      workspaceId: row.workspace_id,
      status: row.status as SessionStatus,
      currentCheckpointId: row.checkpoint_id ?? undefined,
      approvalMode: row.approval_mode as 'auto' | 'semi-auto' | 'manual',
      createdAt: row.created_at,
      updatedAt: row.updated_at,
    };
    
    this.currentSession = session;
    this.sessions.set(session.id, session);
    
    await this.emitter.emit({
      type: 'session.restored',
      timestamp: Date.now(),
      payload: { sessionId },
    });
    
    logger.info('Session restored', { sessionId });
    
    return session;
  }

  async getLastSession(workspaceId: string): Promise<SessionMeta | null> {
    const row = this.db.get<{
      id: string;
      workspace_id: string;
      status: string;
      checkpoint_id: string | null;
      approval_mode: string;
      created_at: number;
      updated_at: number;
    }>(
      'SELECT * FROM sessions WHERE workspace_id = ? ORDER BY updated_at DESC LIMIT 1',
      [workspaceId]
    );
    
    if (!row) return null;
    
    return {
      id: row.id,
      workspaceId: row.workspace_id,
      status: row.status as SessionStatus,
      currentCheckpointId: row.checkpoint_id ?? undefined,
      approvalMode: row.approval_mode as 'auto' | 'semi-auto' | 'manual',
      createdAt: row.created_at,
      updatedAt: row.updated_at,
    };
  }

  getCurrent(): SessionMeta | null {
    return this.currentSession;
  }

  get(sessionId: string): SessionMeta | undefined {
    return this.sessions.get(sessionId);
  }

  async updateTask(sessionId: string, task: TaskSummary): Promise<void> {
    const session = this.sessions.get(sessionId);
    
    if (!session) return;
    
    session.currentTask = task;
    session.updatedAt = Date.now();
    
    this.db.run(
      'UPDATE sessions SET summary = ?, updated_at = ? WHERE id = ?',
      [JSON.stringify(task), session.updatedAt, sessionId]
    );
    
    await this.emitter.emit({
      type: 'session.task-updated',
      timestamp: Date.now(),
      payload: { sessionId, task },
    });
  }

  async setCheckpoint(sessionId: string, checkpointId: string): Promise<void> {
    const session = this.sessions.get(sessionId);
    
    if (!session) return;
    
    session.currentCheckpointId = checkpointId;
    session.updatedAt = Date.now();
    
    this.db.run(
      'UPDATE sessions SET checkpoint_id = ?, updated_at = ? WHERE id = ?',
      [checkpointId, session.updatedAt, sessionId]
    );
  }

  async setApprovalMode(
    sessionId: string,
    mode: 'auto' | 'semi-auto' | 'manual'
  ): Promise<void> {
    const session = this.sessions.get(sessionId);
    
    if (!session) return;
    
    session.approvalMode = mode;
    session.updatedAt = Date.now();
    
    this.db.run(
      'UPDATE sessions SET approval_mode = ?, updated_at = ? WHERE id = ?',
      [mode, session.updatedAt, sessionId]
    );
  }

  async archive(sessionId: string): Promise<void> {
    const session = this.sessions.get(sessionId);
    
    if (!session) return;
    
    session.status = 'archived';
    session.updatedAt = Date.now();
    
    this.db.run(
      'UPDATE sessions SET status = ?, updated_at = ? WHERE id = ?',
      ['archived', session.updatedAt, sessionId]
    );
    
    if (this.currentSession?.id === sessionId) {
      this.currentSession = null;
    }
  }
}
```

```typescript
// src/protection/dirty-check.ts

import { createLogger } from '@ai-digital-life/logger';
import type { GitStatus } from '../types';

const logger = createLogger('dirty-check');

export interface DirtyCheckResult {
  isDirty: boolean;
  reasons: string[];
  canProceed: boolean;
}

export class DirtyChecker {
  async check(gitStatus: GitStatus | undefined): Promise<DirtyCheckResult> {
    const reasons: string[] = [];
    
    if (!gitStatus) {
      return { isDirty: false, reasons: [], canProceed: true };
    }
    
    if (gitStatus.staged.length > 0) {
      reasons.push(`有 ${gitStatus.staged.length} 个已暂存的文件`);
    }
    
    if (gitStatus.unstaged.length > 0) {
      reasons.push(`有 ${gitStatus.unstaged.length} 个未暂存的修改`);
    }
    
    if (gitStatus.untracked.length > 0) {
      reasons.push(`有 ${gitStatus.untracked.length} 个未跟踪的文件`);
    }
    
    if (gitStatus.conflicts.length > 0) {
      reasons.push(`有 ${gitStatus.conflicts.length} 个冲突文件`);
    }
    
    const isDirty = reasons.length > 0;
    
    if (isDirty) {
      logger.warn('Dirty working tree detected', { reasons });
    }
    
    return {
      isDirty,
      reasons,
      canProceed: gitStatus.conflicts.length === 0,
    };
  }

  async checkExternalModification(
    filePath: string,
    lastKnownMtime: number
  ): Promise<boolean> {
    const fs = await import('fs/promises');
    
    try {
      const stats = await fs.stat(filePath);
      return stats.mtimeMs > lastKnownMtime;
    } catch {
      return false;
    }
  }
}
```
