# 任务与审批中心实施计划

## 模块概述

**包名**: `@ai-digital-life/task`

**优先级**: P0 - AI安全执行的核心保障

**目标**: 管理AI执行任务的完整生命周期，包括任务创建、执行跟踪、审批流程和结果记录。

## 设计原则

- **可控性**: 高风险操作必须经过审批
- **可追溯**: 所有任务执行过程可审计
- **可恢复**: 任务失败可回滚到安全状态
- **透明性**: 用户可随时了解任务执行状态

### 任务2: 风险评估器 (1天)

```typescript
// src/approval/risk-assessor.ts

import { createLogger } from '@ai-digital-life/logger';
import type { TaskOperation, RiskLevel } from '../types';

const logger = createLogger('risk-assessor');

export interface RiskAssessment {
  level: RiskLevel;
  score: number;
  factors: string[];
  requiresApproval: boolean;
}

const HIGH_RISK_OPERATIONS = [
  'file.delete',
  'file.batch-delete',
  'command.execute',
  'config.modify',
  'git.force-push',
  'git.reset-hard',
];

const MEDIUM_RISK_OPERATIONS = [
  'file.write',
  'file.batch-write',
  'git.commit',
  'git.push',
  'package.install',
];

const SENSITIVE_PATHS = [
  '.env',
  '.git',
  'credentials',
  'secrets',
  'config',
  'settings',
];

export class RiskAssessor {
  private thresholds: {
    low: number;
    medium: number;
    high: number;
  };

  constructor(thresholds?: { low: number; medium: number; high: number }) {
    this.thresholds = thresholds ?? {
      low: 1,
      medium: 5,
      high: 10,
    };
  }

  assess(operations: TaskOperation[]): RiskAssessment {
    let score = 0;
    const factors: string[] = [];

    for (const op of operations) {
      const opScore = this.assessOperation(op);
      score += opScore;

      if (opScore >= 5) {
        factors.push(`高风险操作: ${op.type}`);
      } else if (opScore >= 2) {
        factors.push(`中等风险操作: ${op.type}`);
      }

      if (op.params.path && this.isSensitivePath(op.params.path as string)) {
        score += 3;
        factors.push(`敏感路径: ${op.params.path}`);
      }
    }

    const level = this.calculateLevel(score);
    const requiresApproval = level !== 'low' || operations.some(op => op.riskLevel !== 'low');

    logger.debug('Risk assessment complete', {
      score,
      level,
      factors,
      requiresApproval,
    });

    return {
      level,
      score,
      factors,
      requiresApproval,
    };
  }

  private assessOperation(operation: TaskOperation): number {
    if (HIGH_RISK_OPERATIONS.includes(operation.type)) {
      return 10;
    }

    if (MEDIUM_RISK_OPERATIONS.includes(operation.type)) {
      return 5;
    }

    if (operation.type.startsWith('file.')) {
      return 2;
    }

    return 1;
  }

  private isSensitivePath(path: string): boolean {
    const lowerPath = path.toLowerCase();
    return SENSITIVE_PATHS.some(sensitive => 
      lowerPath.includes(sensitive.toLowerCase())
    );
  }

  private calculateLevel(score: number): RiskLevel {
    if (score >= this.thresholds.high) {
      return 'high';
    }
    if (score >= this.thresholds.medium) {
      return 'medium';
    }
    return 'low';
  }
}
```

### 任务3: 审批管理器 (2天)

```typescript
// src/approval/approval-manager.ts

import { generateId } from '@ai-digital-life/kernel';
import { EventEmitter } from '@ai-digital-life/events';
import { createLogger } from '@ai-digital-life/logger';
import type { DatabaseConnection } from '@ai-digital-life/storage';
import { RiskAssessor } from './risk-assessor';
import type { 
  ApprovalRequest, 
  Approval, 
  ApprovalMode, 
  ApprovalDecision,
  TaskOperation 
} from '../types';

const logger = createLogger('approval-manager');

export class ApprovalManager {
  private db: DatabaseConnection;
  private emitter: EventEmitter;
  private assessor: RiskAssessor;
  private mode: ApprovalMode;
  private pendingRequests = new Map<string, ApprovalRequest>();

  constructor(
    db: DatabaseConnection,
    emitter: EventEmitter,
    mode: ApprovalMode = 'semi-auto'
  ) {
    this.db = db;
    this.emitter = emitter;
    this.assessor = new RiskAssessor();
    this.mode = mode;
  }

  setMode(mode: ApprovalMode): void {
    this.mode = mode;
    logger.info('Approval mode changed', { mode });
  }

  async requestApproval(
    taskId: string,
    workspaceId: string,
    operations: TaskOperation[]
  ): Promise<{ requiresApproval: boolean; requestId?: string }> {
    const assessment = this.assessor.assess(operations);

    if (this.shouldAutoApprove(assessment.level)) {
      logger.info('Auto-approving task', { taskId, riskLevel: assessment.level });
      return { requiresApproval: false };
    }

    const request = await this.createRequest(taskId, workspaceId, operations, assessment);
    this.pendingRequests.set(request.id, request);

    await this.emitter.emit({
      type: 'approval.requested',
      timestamp: Date.now(),
      payload: {
        approvalId: request.id,
        taskId,
        riskLevel: request.riskLevel,
        description: request.description,
        operations: request.operations,
      },
    });

    logger.info('Approval requested', { 
      requestId: request.id, 
      taskId, 
      riskLevel: request.riskLevel 
    });

    return { requiresApproval: true, requestId: request.id };
  }

  private shouldAutoApprove(riskLevel: string): boolean {
    if (this.mode === 'auto') return true;
    if (this.mode === 'manual') return false;
    return riskLevel === 'low';
  }

  private async createRequest(
    taskId: string,
    workspaceId: string,
    operations: TaskOperation[],
    assessment: { level: string; score: number; factors: string[] }
  ): Promise<ApprovalRequest> {
    const id = generateId();
    
    const request: ApprovalRequest = {
      id,
      taskId,
      workspaceId,
      riskLevel: assessment.level as 'low' | 'medium' | 'high',
      title: this.generateTitle(operations),
      description: this.generateDescription(operations, assessment),
      operations,
      impact: assessment.factors,
      reversible: operations.every(op => op.reversible),
      createdAt: Date.now(),
      expiresAt: Date.now() + 5 * 60 * 1000, // 5分钟过期
    };

    this.db.run(
      `INSERT INTO approvals (id, task_id, risk_level, decision, created_at)
       VALUES (?, ?, ?, NULL, ?)`,
      [id, taskId, request.riskLevel, request.createdAt]
    );

    return request;
  }

  private generateTitle(operations: TaskOperation[]): string {
    if (operations.length === 1) {
      return `执行操作: ${operations[0].type}`;
    }
    return `执行 ${operations.length} 个操作`;
  }

  private generateDescription(
    operations: TaskOperation[],
    assessment: { level: string; score: number; factors: string[] }
  ): string {
    const lines: string[] = [];
    
    lines.push(`风险等级: ${assessment.level}`);
    lines.push(`风险评分: ${assessment.score}`);
    lines.push('');
    lines.push('操作列表:');
    
    for (const op of operations) {
      lines.push(`- ${op.type}: ${JSON.stringify(op.params)}`);
    }
    
    if (assessment.factors.length > 0) {
      lines.push('');
      lines.push('风险因素:');
      for (const factor of assessment.factors) {
        lines.push(`- ${factor}`);
      }
    }
    
    return lines.join('\n');
  }

  async approve(
    requestId: string,
    approvedBy: string,
    reason?: string
  ): Promise<Approval> {
    const request = this.pendingRequests.get(requestId);
    
    if (!request) {
      throw new Error(`Approval request not found: ${requestId}`);
    }

    const approval: Approval = {
      id: generateId(),
      requestId,
      taskId: request.taskId,
      decision: 'approved',
      reason,
      approvedBy,
      approvedAt: Date.now(),
      createdAt: Date.now(),
    };

    this.db.run(
      `UPDATE approvals SET decision = ?, reason = ?, decided_at = ? WHERE id = ?`,
      ['approved', reason ?? null, approval.approvedAt, requestId]
    );

    this.pendingRequests.delete(requestId);

    await this.emitter.emit({
      type: 'approval.decided',
      timestamp: Date.now(),
      payload: {
        approvalId: requestId,
        decision: 'approved',
        reason,
      },
    });

    logger.info('Approval granted', { requestId, approvedBy });

    return approval;
  }

  async reject(
    requestId: string,
    reason: string
  ): Promise<Approval> {
    const request = this.pendingRequests.get(requestId);
    
    if (!request) {
      throw new Error(`Approval request not found: ${requestId}`);
    }

    const approval: Approval = {
      id: generateId(),
      requestId,
      taskId: request.taskId,
      decision: 'rejected',
      reason,
      createdAt: Date.now(),
    };

    this.db.run(
      `UPDATE approvals SET decision = ?, reason = ?, decided_at = ? WHERE id = ?`,
      ['rejected', reason, Date.now(), requestId]
    );

    this.pendingRequests.delete(requestId);

    await this.emitter.emit({
      type: 'approval.decided',
      timestamp: Date.now(),
      payload: {
        approvalId: requestId,
        decision: 'rejected',
        reason,
      },
    });

    logger.info('Approval rejected', { requestId, reason });

    return approval;
  }

  async getPendingRequests(): Promise<ApprovalRequest[]> {
    return Array.from(this.pendingRequests.values());
  }

  async getRequest(requestId: string): Promise<ApprovalRequest | undefined> {
    return this.pendingRequests.get(requestId);
  }

  async checkExpiration(): Promise<string[]> {
    const now = Date.now();
    const expired: string[] = [];

    for (const [id, request] of this.pendingRequests) {
      if (request.expiresAt && request.expiresAt < now) {
        await this.timeout(id);
        expired.push(id);
      }
    }

    return expired;
  }

  private async timeout(requestId: string): Promise<void> {
    const request = this.pendingRequests.get(requestId);
    
    if (!request) return;

    this.db.run(
      `UPDATE approvals SET decision = ?, decided_at = ? WHERE id = ?`,
      ['timeout', Date.now(), requestId]
    );

    this.pendingRequests.delete(requestId);

    await this.emitter.emit({
      type: 'approval.decided',
      timestamp: Date.now(),
      payload: {
        approvalId: requestId,
        decision: 'timeout',
      },
    });

    logger.warn('Approval request timed out', { requestId });
  }
}
```

```typescript
// src/task/task-manager.ts

import { generateId } from '@ai-digital-life/kernel';
import { EventEmitter } from '@ai-digital-life/events';
import { createLogger } from '@ai-digital-life/logger';
import type { DatabaseConnection } from '@ai-digital-life/storage';
import { ApprovalManager } from '../approval/approval-manager';
import type { Task, TaskStatus, TaskOperation, TaskPriority } from '../types';

const logger = createLogger('task-manager');

export class TaskManager {
  private db: DatabaseConnection;
  private emitter: EventEmitter;
  private approvalManager: ApprovalManager;
  private tasks = new Map<string, Task>();

  constructor(
    db: DatabaseConnection,
    emitter: EventEmitter,
    approvalManager: ApprovalManager
  ) {
    this.db = db;
    this.emitter = emitter;
    this.approvalManager = approvalManager;
  }

  async create(
    workspaceId: string,
    sessionId: string,
    type: string,
    description: string,
    operations: TaskOperation[],
    options: {
      priority?: TaskPriority;
      parentId?: string;
    } = {}
  ): Promise<Task> {
    const id = generateId();
    
    const task: Task = {
      id,
      workspaceId,
      sessionId,
      parentId: options.parentId,
      type,
      status: 'pending',
      priority: options.priority ?? 'medium',
      description,
      operations,
      createdAt: Date.now(),
      updatedAt: Date.now(),
    };

    this.db.run(
      `INSERT INTO tasks (id, workspace_id, session_id, type, status, priority, description, created_at, updated_at)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [id, workspaceId, sessionId, type, 'pending', task.priority, description, task.createdAt, task.updatedAt]
    );

    this.tasks.set(id, task);

    await this.emitter.emit({
      type: 'task.created',
      timestamp: Date.now(),
      payload: { taskId: id, type, description },
    });

    logger.info('Task created', { taskId: id, type, description });

    return task;
  }

  async start(taskId: string): Promise<void> {
    const task = this.tasks.get(taskId);
    
    if (!task) {
      throw new Error(`Task not found: ${taskId}`);
    }

    if (task.status !== 'pending') {
      throw new Error(`Task cannot be started: ${task.status}`);
    }

    const { requiresApproval, requestId } = await this.approvalManager.requestApproval(
      taskId,
      task.workspaceId,
      task.operations
    );

    if (requiresApproval && requestId) {
      await this.updateStatus(taskId, 'waiting-approval');
      task.approvalId = requestId;
      return;
    }

    await this.updateStatus(taskId, 'running');
    task.startedAt = Date.now();
  }

  async complete(taskId: string, result: unknown): Promise<void> {
    const task = this.tasks.get(taskId);
    
    if (!task) {
      throw new Error(`Task not found: ${taskId}`);
    }

    task.result = result;
    task.completedAt = Date.now();

    await this.updateStatus(taskId, 'completed');

    await this.emitter.emit({
      type: 'task.completed',
      timestamp: Date.now(),
      payload: { taskId, result },
    });

    logger.info('Task completed', { taskId });
  }

  async fail(taskId: string, error: { code: string; message: string; details?: Record<string, unknown> }): Promise<void> {
    const task = this.tasks.get(taskId);
    
    if (!task) {
      throw new Error(`Task not found: ${taskId}`);
    }

    task.error = {
      ...error,
      recoverable: this.isRecoverable(error),
    };

    await this.updateStatus(taskId, 'failed');

    await this.emitter.emit({
      type: 'task.failed',
      timestamp: Date.now(),
      payload: { taskId, error },
    });

    logger.error('Task failed', new Error(error.message), { taskId, error });
  }

  async rollback(taskId: string): Promise<void> {
    const task = this.tasks.get(taskId);
    
    if (!task) {
      throw new Error(`Task not found: ${taskId}`);
    }

    // 实际回滚逻辑需要根据任务类型实现
    await this.updateStatus(taskId, 'rolled-back');

    logger.info('Task rolled back', { taskId });
  }

  private async updateStatus(taskId: string, status: TaskStatus): Promise<void> {
    const task = this.tasks.get(taskId);
    
    if (!task) return;

    task.status = status;
    task.updatedAt = Date.now();

    this.db.run(
      'UPDATE tasks SET status = ?, updated_at = ? WHERE id = ?',
      [status, task.updatedAt, taskId]
    );

    await this.emitter.emit({
      type: 'task.status-changed',
      timestamp: Date.now(),
      payload: { taskId, status },
    });
  }

  private isRecoverable(error: { code: string; message: string }): boolean {
    const recoverableCodes = ['E2001', 'E2002', 'E3002'];
    return recoverableCodes.includes(error.code);
  }

  get(taskId: string): Task | undefined {
    return this.tasks.get(taskId);
  }

  getBySession(sessionId: string): Task[] {
    return Array.from(this.tasks.values()).filter(t => t.sessionId === sessionId);
  }

  getByWorkspace(workspaceId: string): Task[] {
    return Array.from(this.tasks.values()).filter(t => t.workspaceId === workspaceId);
  }

  async loadTask(taskId: string): Promise<Task | undefined> {
    const row = this.db.get<{
      id: string;
      workspace_id: string;
      session_id: string;
      type: string;
      status: string;
      description: string;
      result: string | null;
      error: string | null;
      created_at: number;
      updated_at: number;
      started_at: number | null;
      completed_at: number | null;
    }>('SELECT * FROM tasks WHERE id = ?', [taskId]);

    if (!row) return undefined;

    return {
      id: row.id,
      workspaceId: row.workspace_id,
      sessionId: row.session_id,
      type: row.type,
      status: row.status as TaskStatus,
      priority: 'medium',
      description: row.description,
      operations: [],
      result: row.result ? JSON.parse(row.result) : undefined,
      error: row.error ? JSON.parse(row.error) : undefined,
      createdAt: row.created_at,
      updatedAt: row.updated_at,
      startedAt: row.started_at ?? undefined,
      completedAt: row.completed_at ?? undefined,
    };
  }
}
```

```typescript
// src/history/task-history.ts

import { createLogger } from '@ai-digital-life/logger';
import type { DatabaseConnection } from '@ai-digital-life/storage';
import type { Task, TaskStatus } from '../types';

const logger = createLogger('task-history');

export interface TaskHistoryEntry {
  taskId: string;
  status: TaskStatus;
  timestamp: number;
  details?: Record<string, unknown>;
}

export class TaskHistory {
  private db: DatabaseConnection;

  constructor(db: DatabaseConnection) {
    this.db = db;
  }

  async record(entry: TaskHistoryEntry): Promise<void> {
    this.db.run(
      `INSERT INTO task_history (task_id, status, timestamp, details)
       VALUES (?, ?, ?, ?)`,
      [entry.taskId, entry.status, entry.timestamp, JSON.stringify(entry.details ?? {})]
    );
  }

  async getTaskHistory(taskId: string): Promise<TaskHistoryEntry[]> {
    const rows = this.db.all<{
      task_id: string;
      status: string;
      timestamp: number;
      details: string;
    }>(
      'SELECT * FROM task_history WHERE task_id = ? ORDER BY timestamp ASC',
      [taskId]
    );

    return rows.map(row => ({
      taskId: row.task_id,
      status: row.status as TaskStatus,
      timestamp: row.timestamp,
      details: JSON.parse(row.details),
    }));
  }

  async getRecentTasks(workspaceId: string, limit = 50): Promise<Task[]> {
    const rows = this.db.all<{
      id: string;
      workspace_id: string;
      session_id: string;
      type: string;
      status: string;
      description: string;
      created_at: number;
      completed_at: number | null;
    }>(
      `SELECT * FROM tasks 
       WHERE workspace_id = ? 
       ORDER BY created_at DESC 
       LIMIT ?`,
      [workspaceId, limit]
    );

    return rows.map(row => ({
      id: row.id,
      workspaceId: row.workspace_id,
      sessionId: row.session_id,
      type: row.type,
      status: row.status as TaskStatus,
      priority: 'medium' as const,
      description: row.description,
      operations: [],
      createdAt: row.created_at,
      updatedAt: row.created_at,
      completedAt: row.completed_at ?? undefined,
    }));
  }

  async getStatistics(workspaceId: string): Promise<{
    total: number;
    completed: number;
    failed: number;
    avgDuration: number;
  }> {
    const total = this.db.get<{ count: number }>(
      'SELECT COUNT(*) as count FROM tasks WHERE workspace_id = ?',
      [workspaceId]
    )?.count ?? 0;

    const completed = this.db.get<{ count: number }>(
      'SELECT COUNT(*) as count FROM tasks WHERE workspace_id = ? AND status = ?',
      [workspaceId, 'completed']
    )?.count ?? 0;

    const failed = this.db.get<{ count: number }>(
      'SELECT COUNT(*) as count FROM tasks WHERE workspace_id = ? AND status = ?',
      [workspaceId, 'failed']
    )?.count ?? 0;

    const avgDuration = this.db.get<{ avg: number }>(
      `SELECT AVG(completed_at - created_at) as avg 
       FROM tasks 
       WHERE workspace_id = ? AND completed_at IS NOT NULL`,
      [workspaceId]
    )?.avg ?? 0;

    return { total, completed, failed, avgDuration };
  }
}
```
