# 日志与监控系统实施计划

## 模块概述

**包名**: `@ai-digital-life/logger`

**优先级**: P0 - 问题排查和审计的基础设施

**目标**: 为应用提供结构化日志记录、性能监控和运行时诊断能力。

## 设计原则

- **结构化**: 所有日志采用结构化格式，便于查询和分析
- **多级别**: 支持DEBUG/INFO/WARN/ERROR等级别
- **高性能**: 日志写入不阻塞主线程
- **可配置**: 日志级别、输出目标可动态配置

```typescript
// src/types/level.ts

export enum LogLevel {
  DEBUG = 0,
  INFO = 1,
  WARN = 2,
  ERROR = 3,
  SILENT = 4,
}

export const LOG_LEVEL_NAMES: Record<LogLevel, string> = {
  [LogLevel.DEBUG]: 'DEBUG',
  [LogLevel.INFO]: 'INFO',
  [LogLevel.WARN]: 'WARN',
  [LogLevel.ERROR]: 'ERROR',
  [LogLevel.SILENT]: 'SILENT',
};

export function parseLogLevel(level: string): LogLevel {
  const upperLevel = level.toUpperCase();
  
  for (const [value, name] of Object.entries(LOG_LEVEL_NAMES)) {
    if (name === upperLevel) {
      return parseInt(value);
    }
  }
  
  return LogLevel.INFO;
}
```

```typescript
// src/transports/console.ts

import type { LogEntry } from '../types';
import { LogLevel } from '../types';

export interface ConsoleTransportOptions {
  colorize?: boolean;
  format?: 'json' | 'pretty';
}

const COLORS = {
  [LogLevel.DEBUG]: '\x1b[36m', // cyan
  [LogLevel.INFO]: '\x1b[32m',  // green
  [LogLevel.WARN]: '\x1b[33m',  // yellow
  [LogLevel.ERROR]: '\x1b[31m', // red
  reset: '\x1b[0m',
};

export class ConsoleTransport {
  private options: ConsoleTransportOptions;

  constructor(options: ConsoleTransportOptions = {}) {
    this.options = {
      colorize: true,
      format: 'pretty',
      ...options,
    };
  }

  write(entry: LogEntry): void {
    const output = this.options.format === 'json'
      ? this.formatJson(entry)
      : this.formatPretty(entry);

    const method = this.getConsoleMethod(entry.level);
    method(output);
  }

  private formatJson(entry: LogEntry): string {
    return JSON.stringify({
      timestamp: new Date(entry.timestamp).toISOString(),
      level: entry.levelName,
      module: entry.module,
      message: entry.message,
      ...entry.context,
      ...(entry.error && { error: entry.error }),
      ...(entry.duration !== undefined && { duration: entry.duration }),
    });
  }

  private formatPretty(entry: LogEntry): string {
    const time = new Date(entry.timestamp).toISOString();
    const level = entry.levelName.padEnd(5);
    
    let output = `[${time}] ${level} [${entry.module}] ${entry.message}`;
    
    if (entry.context && Object.keys(entry.context).length > 0) {
      output += ` ${JSON.stringify(entry.context)}`;
    }
    
    if (entry.duration !== undefined) {
      output += ` (${entry.duration}ms)`;
    }
    
    if (entry.error) {
      output += `\n  ${entry.error.name}: ${entry.error.message}`;
      if (entry.error.stack) {
        output += `\n  ${entry.error.stack}`;
      }
    }
    
    if (this.options.colorize) {
      const color = COLORS[entry.level] ?? '';
      output = `${color}${output}${COLORS.reset}`;
    }
    
    return output;
  }

  private getConsoleMethod(level: number): (...args: unknown[]) => void {
    switch (level) {
      case LogLevel.ERROR:
        return console.error;
      case LogLevel.WARN:
        return console.warn;
      case LogLevel.DEBUG:
        return console.debug;
      default:
        return console.log;
    }
  }
}
```

```typescript
// src/transports/file.ts

import fs from 'fs/promises';
import path from 'path';
import type { LogEntry } from '../types';

export interface FileTransportOptions {
  logDir: string;
  filename?: string;
  maxSize?: number;      // 最大文件大小（字节）
  maxFiles?: number;     // 最大文件数量
  compress?: boolean;    // 是否压缩旧日志
}

export class FileTransport {
  private options: Required<FileTransportOptions>;
  private currentFile: string | null = null;
  private currentSize = 0;
  private writeQueue: LogEntry[] = [];
  private isWriting = false;

  constructor(options: FileTransportOptions) {
    this.options = {
      logDir: options.logDir,
      filename: options.filename ?? 'app.log',
      maxSize: options.maxSize ?? 10 * 1024 * 1024, // 10MB
      maxFiles: options.maxFiles ?? 30,
      compress: options.compress ?? true,
    };
  }

  async initialize(): Promise<void> {
    await fs.mkdir(this.options.logDir, { recursive: true });
    this.currentFile = path.join(this.options.logDir, this.options.filename);
    
    try {
      const stats = await fs.stat(this.currentFile);
      this.currentSize = stats.size;
    } catch {
      this.currentSize = 0;
    }
  }

  write(entry: LogEntry): void {
    this.writeQueue.push(entry);
    this.processQueue();
  }

  private async processQueue(): Promise<void> {
    if (this.isWriting || this.writeQueue.length === 0) return;
    
    this.isWriting = true;
    
    try {
      while (this.writeQueue.length > 0) {
        const entry = this.writeQueue.shift()!;
        await this.writeEntry(entry);
      }
    } finally {
      this.isWriting = false;
    }
  }

  private async writeEntry(entry: LogEntry): Promise<void> {
    const line = this.formatLine(entry);
    
    if (this.currentSize + line.length > this.options.maxSize) {
      await this.rotate();
    }
    
    await fs.appendFile(this.currentFile!, line);
    this.currentSize += line.length;
  }

  private formatLine(entry: LogEntry): string {
    return JSON.stringify({
      timestamp: new Date(entry.timestamp).toISOString(),
      level: entry.levelName,
      module: entry.module,
      message: entry.message,
      ...entry.context,
      ...(entry.error && { error: entry.error }),
      ...(entry.duration !== undefined && { duration: entry.duration }),
    }) + '\n';
  }

  private async rotate(): Promise<void> {
    if (!this.currentFile) return;
    
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const rotatedFile = this.currentFile.replace('.log', `.${timestamp}.log`);
    
    await fs.rename(this.currentFile, rotatedFile);
    this.currentSize = 0;
    
    if (this.options.compress) {
      await this.compressFile(rotatedFile);
    }
    
    await this.cleanOldFiles();
  }

  private async compressFile(filepath: string): Promise<void> {
    const { createGzip } = await import('zlib');
    const { pipeline } = await import('stream/promises');
    const { createReadStream, createWriteStream } = await import('fs');
    
    const gzPath = `${filepath}.gz`;
    
    await pipeline(
      createReadStream(filepath),
      createGzip(),
      createWriteStream(gzPath)
    );
    
    await fs.unlink(filepath);
  }

  private async cleanOldFiles(): Promise<void> {
    const files = await fs.readdir(this.options.logDir);
    const logFiles = files
      .filter(f => f.endsWith('.log') || f.endsWith('.log.gz'))
      .sort()
      .reverse();
    
    while (logFiles.length > this.options.maxFiles) {
      const toDelete = logFiles.pop()!;
      await fs.unlink(path.join(this.options.logDir, toDelete));
    }
  }
}
```

```typescript
// src/logger.ts

import { generateId } from '@ai-digital-life/kernel';
import { LogLevel, parseLogLevel, type LogEntry, type LogContext } from './types';
import { ConsoleTransport } from './transports/console';
import type { FileTransport } from './transports/file';

export interface LoggerOptions {
  level?: LogLevel | string;
  module?: string;
  context?: LogContext;
}

export class Logger {
  private static globalLevel: LogLevel = LogLevel.INFO;
  private static transports: Array<{ write: (entry: LogEntry) => void }> = [];
  private static initialized = false;

  private module: string;
  private context: LogContext;

  constructor(module: string, context: LogContext = {}) {
    this.module = module;
    this.context = context;
  }

  static initialize(options: {
    level?: string;
    logDir?: string;
    console?: boolean;
  } = {}): void {
    if (Logger.initialized) return;

    if (options.level) {
      Logger.globalLevel = parseLogLevel(options.level);
    }

    if (options.console !== false) {
      Logger.transports.push(new ConsoleTransport());
    }

    if (options.logDir) {
      const fileTransport = new FileTransport({ logDir: options.logDir });
      fileTransport.initialize().catch(console.error);
      Logger.transports.push(fileTransport);
    }

    Logger.initialized = true;
  }

  static setLevel(level: LogLevel | string): void {
    Logger.globalLevel = typeof level === 'string' ? parseLogLevel(level) : level;
  }

  static addTransport(transport: { write: (entry: LogEntry) => void }): void {
    Logger.transports.push(transport);
  }

  child(module: string, context: LogContext = {}): Logger {
    return new Logger(
      `${this.module}:${module}`,
      { ...this.context, ...context }
    );
  }

  debug(message: string, context?: Record<string, unknown>): void {
    this.log(LogLevel.DEBUG, message, context);
  }

  info(message: string, context?: Record<string, unknown>): void {
    this.log(LogLevel.INFO, message, context);
  }

  warn(message: string, context?: Record<string, unknown>): void {
    this.log(LogLevel.WARN, message, context);
  }

  error(message: string, error?: Error | unknown, context?: Record<string, unknown>): void {
    const entry: LogEntry = this.createEntry(LogLevel.ERROR, message, context);
    
    if (error instanceof Error) {
      entry.error = {
        name: error.name,
        message: error.message,
        stack: error.stack,
      };
    } else if (error) {
      entry.context = { ...entry.context, error };
    }
    
    this.write(entry);
  }

  time(label?: string): () => number {
    const start = Date.now();
    const id = label ?? generateId();
    
    this.debug(`Timer started: ${id}`);
    
    return () => {
      const duration = Date.now() - start;
      this.debug(`Timer ended: ${id}`, { duration });
      return duration;
    };
  }

  async measure<T>(label: string, fn: () => Promise<T>): Promise<T> {
    const start = Date.now();
    
    try {
      const result = await fn();
      const duration = Date.now() - start;
      this.debug(label, { duration, success: true });
      return result;
    } catch (error) {
      const duration = Date.now() - start;
      this.error(label, error, { duration, success: false });
      throw error;
    }
  }

  private log(level: LogLevel, message: string, context?: Record<string, unknown>): void {
    if (level < Logger.globalLevel) return;
    
    const entry = this.createEntry(level, message, context);
    this.write(entry);
  }

  private createEntry(
    level: LogLevel,
    message: string,
    context?: Record<string, unknown>
  ): LogEntry {
    return {
      timestamp: Date.now(),
      level,
      levelName: LogLevel[level] as string,
      module: this.module,
      message,
      context: { ...this.context, ...context },
    };
  }

  private write(entry: LogEntry): void {
    for (const transport of Logger.transports) {
      try {
        transport.write(entry);
      } catch (error) {
        console.error('Logger transport error:', error);
      }
    }
  }
}

export function createLogger(module: string, context?: LogContext): Logger {
  return new Logger(module, context);
}
```

### 任务4: 性能监控 (1天)

```typescript
// src/performance.ts

import { performance, PerformanceObserver } from 'perf_hooks';

export interface PerformanceMetric {
  name: string;
  value: number;
  unit: 'ms' | 'bytes' | 'count';
  timestamp: number;
  tags?: Record<string, string>;
}

export class PerformanceMonitor {
  private metrics: PerformanceMetric[] = [];
  private observer: PerformanceObserver | null = null;
  private maxMetrics = 10000;

  start(): void {
    this.observer = new PerformanceObserver(list => {
      for (const entry of list.getEntries()) {
        this.record({
          name: entry.name,
          value: entry.duration,
          unit: 'ms',
          timestamp: entry.startTime,
        });
      }
    });
    
    this.observer.observe({ entryTypes: ['measure'] });
  }

  stop(): void {
    this.observer?.disconnect();
    this.observer = null;
  }

  record(metric: PerformanceMetric): void {
    this.metrics.push(metric);
    
    if (this.metrics.length > this.maxMetrics) {
      this.metrics.shift();
    }
  }

  measure(name: string, fn: () => void): number {
    const start = performance.now();
    fn();
    const duration = performance.now() - start;
    
    this.record({
      name,
      value: duration,
      unit: 'ms',
      timestamp: Date.now(),
    });
    
    return duration;
  }

  async measureAsync<T>(name: string, fn: () => Promise<T>): Promise<{ result: T; duration: number }> {
    const start = performance.now();
    const result = await fn();
    const duration = performance.now() - start;
    
    this.record({
      name,
      value: duration,
      unit: 'ms',
      timestamp: Date.now(),
    });
    
    return { result, duration };
  }

  getMetrics(name?: string): PerformanceMetric[] {
    if (name) {
      return this.metrics.filter(m => m.name === name);
    }
    return [...this.metrics];
  }

  getStats(name: string): {
    count: number;
    min: number;
    max: number;
    avg: number;
    p50: number;
    p95: number;
    p99: number;
  } {
    const values = this.metrics
      .filter(m => m.name === name)
      .map(m => m.value)
      .sort((a, b) => a - b);

    if (values.length === 0) {
      return { count: 0, min: 0, max: 0, avg: 0, p50: 0, p95: 0, p99: 0 };
    }

    const sum = values.reduce((a, b) => a + b, 0);
    const count = values.length;

    return {
      count,
      min: values[0],
      max: values[count - 1],
      avg: sum / count,
      p50: values[Math.floor(count * 0.5)],
      p95: values[Math.floor(count * 0.95)],
      p99: values[Math.floor(count * 0.99)],
    };
  }

  getMemoryUsage(): NodeJS.MemoryUsage {
    return process.memoryUsage();
  }

  getCpuUsage(): NodeJS.CpuUsage {
    return process.cpuUsage();
  }

  clear(): void {
    this.metrics = [];
  }
}

export const performanceMonitor = new PerformanceMonitor();
```

```typescript
// tests/logger.test.ts

import { describe, it, expect, beforeEach, vi } from 'vitest';
import { Logger, LogLevel } from '../src/logger';

describe('Logger', () => {
  let logger: Logger;
  let mockTransport: { write: ReturnType<typeof vi.fn> };

  beforeEach(() => {
    mockTransport = { write: vi.fn() };
    Logger.transports = [mockTransport];
    Logger.globalLevel = LogLevel.DEBUG;
    logger = new Logger('test');
  });

  it('should log debug messages', () => {
    logger.debug('test message');
    expect(mockTransport.write).toHaveBeenCalledWith(
      expect.objectContaining({
        level: LogLevel.DEBUG,
        message: 'test message',
      })
    );
  });

  it('should respect log level', () => {
    Logger.globalLevel = LogLevel.WARN;
    logger.debug('should not appear');
    expect(mockTransport.write).not.toHaveBeenCalled();
  });

  it('should include context', () => {
    logger.info('test', { key: 'value' });
    expect(mockTransport.write).toHaveBeenCalledWith(
      expect.objectContaining({
        context: { key: 'value' },
      })
    );
  });

  it('should format errors', () => {
    const error = new Error('test error');
    logger.error('error occurred', error);
    
    expect(mockTransport.write).toHaveBeenCalledWith(
      expect.objectContaining({
        error: expect.objectContaining({
          message: 'test error',
        }),
      })
    );
  });
});
```
