# 事件系统与消息总线实施计划

## 模块概述

**包名**: `@ai-digital-life/events`

**优先级**: P0 - 模块间通信的核心基础设施

**目标**: 为应用提供模块间松耦合的通信机制，基于发布/订阅模式，支持事件类型定义、事件过滤和优先级队列。

## 设计原则

- **松耦合**: 发布者与订阅者互不感知
- **类型安全**: 事件类型与载荷在编译期校验
- **高性能**: 事件分发延迟 < 0.1ms
- **简洁**: MVP阶段只做核心功能

### 任务2: 优先级队列 (1天)

```typescript
// src/priority-queue.ts

import type { Subscription } from './types';

export class PriorityQueue {
  private items: Subscription[] = [];

  enqueue(item: Subscription): void {
    let added = false;
    
    for (let i = 0; i < this.items.length; i++) {
      if (item.priority > this.items[i].priority) {
        this.items.splice(i, 0, item);
        added = true;
        break;
      }
    }
    
    if (!added) {
      this.items.push(item);
    }
  }

  dequeue(): Subscription | undefined {
    return this.items.shift();
  }

  peek(): Subscription | undefined {
    return this.items[0];
  }

  remove(id: string): boolean {
    const index = this.items.findIndex(item => item.id === id);
    if (index !== -1) {
      this.items.splice(index, 1);
      return true;
    }
    return false;
  }

  getByType(eventType: string): Subscription[] {
    return this.items.filter(item => item.eventType === eventType);
  }

  clear(): void {
    this.items = [];
  }

  get length(): number {
    return this.items.length;
  }
}
```

### 任务3: 事件发射器 (2天)

```typescript
// src/emitter.ts

import { generateId } from '@ai-digital-life/kernel';
import type { BaseEvent, EventHandler, EventOptions, Subscription } from './types';
import { PriorityQueue } from './priority-queue';

export class EventEmitter {
  private subscriptions = new Map<string, PriorityQueue>();
  private onceHandlers = new Set<string>();

  on<T extends BaseEvent>(
    eventType: string,
    handler: EventHandler<T>,
    options: EventOptions = {}
  ): string {
    const id = generateId();
    const subscription: Subscription = {
      id,
      eventType,
      handler: handler as EventHandler,
      priority: options.priority ?? 10,
      once: options.once ?? false,
    };

    if (!this.subscriptions.has(eventType)) {
      this.subscriptions.set(eventType, new PriorityQueue());
    }

    this.subscriptions.get(eventType)!.enqueue(subscription);

    if (options.signal) {
      options.signal.addEventListener('abort', () => {
        this.off(id);
      });
    }

    return id;
  }

  once<T extends BaseEvent>(
    eventType: string,
    handler: EventHandler<T>,
    options: Omit<EventOptions, 'once'> = {}
  ): string {
    return this.on(eventType, handler, { ...options, once: true });
  }

  off(subscriptionId: string): boolean {
    for (const [eventType, queue] of this.subscriptions) {
      if (queue.remove(subscriptionId)) {
        if (queue.length === 0) {
          this.subscriptions.delete(eventType);
        }
        return true;
      }
    }
    return false;
  }

  async emit<T extends BaseEvent>(event: T): Promise<void> {
    const queue = this.subscriptions.get(event.type);
    if (!queue) return;

    const handlers = [...queue.getByType(event.type)];
    const toRemove: string[] = [];

    for (const subscription of handlers) {
      try {
        await subscription.handler(event);
        
        if (subscription.once) {
          toRemove.push(subscription.id);
        }
      } catch (error) {
        console.error(`Error in event handler for ${event.type}:`, error);
      }
    }

    for (const id of toRemove) {
      this.off(id);
    }
  }

  emitSync<T extends BaseEvent>(event: T): void {
    const queue = this.subscriptions.get(event.type);
    if (!queue) return;

    const handlers = [...queue.getByType(event.type)];
    const toRemove: string[] = [];

    for (const subscription of handlers) {
      try {
        const result = subscription.handler(event);
        if (result instanceof Promise) {
          console.warn(`Async handler detected in sync emit for ${event.type}`);
        }
        
        if (subscription.once) {
          toRemove.push(subscription.id);
        }
      } catch (error) {
        console.error(`Error in event handler for ${event.type}:`, error);
      }
    }

    for (const id of toRemove) {
      this.off(id);
    }
  }

  clear(eventType?: string): void {
    if (eventType) {
      this.subscriptions.delete(eventType);
    } else {
      this.subscriptions.clear();
    }
  }

  listenerCount(eventType: string): number {
    const queue = this.subscriptions.get(eventType);
    return queue?.length ?? 0;
  }
}
```

```typescript
// src/bus.ts

import { EventEmitter } from './emitter';
import type { BaseEvent, EventHandler, EventOptions } from './types';

export interface EventBus {
  on<T extends BaseEvent>(
    eventType: string,
    handler: EventHandler<T>,
    options?: EventOptions
  ): string;

  once<T extends BaseEvent>(
    eventType: string,
    handler: EventHandler<T>,
    options?: Omit<EventOptions, 'once'>
  ): string;

  off(subscriptionId: string): boolean;

  emit<T extends BaseEvent>(event: T): Promise<void>;

  emitSync<T extends BaseEvent>(event: T): void;

  clear(eventType?: string): void;

  listenerCount(eventType: string): number;
}

let globalBus: EventBus | null = null;

export function createEventBus(): EventBus {
  return new EventEmitter();
}

export function getGlobalEventBus(): EventBus {
  if (!globalBus) {
    globalBus = createEventBus();
  }
  return globalBus;
}

export function resetGlobalEventBus(): void {
  globalBus = null;
}
```

## 测试计划

```typescript
// tests/emitter.test.ts

import { describe, it, expect, vi, beforeEach } from 'vitest';
import { EventEmitter } from '../src/emitter';

describe('EventEmitter', () => {
  let emitter: EventEmitter;

  beforeEach(() => {
    emitter = new EventEmitter();
  });

  describe('on', () => {
    it('should register event handler', () => {
      const handler = vi.fn();
      emitter.on('test', handler);
      expect(emitter.listenerCount('test')).toBe(1);
    });

    it('should call handler when event is emitted', async () => {
      const handler = vi.fn();
      emitter.on('test', handler);
      await emitter.emit({ type: 'test', timestamp: Date.now() });
      expect(handler).toHaveBeenCalled();
    });
  });

  describe('once', () => {
    it('should only call handler once', async () => {
      const handler = vi.fn();
      emitter.once('test', handler);
      
      await emitter.emit({ type: 'test', timestamp: Date.now() });
      await emitter.emit({ type: 'test', timestamp: Date.now() });
      
      expect(handler).toHaveBeenCalledTimes(1);
      expect(emitter.listenerCount('test')).toBe(0);
    });
  });

  describe('priority', () => {
    it('should call handlers in priority order', async () => {
      const order: number[] = [];
      
      emitter.on('test', () => order.push(1), { priority: 10 });
      emitter.on('test', () => order.push(2), { priority: 100 });
      emitter.on('test', () => order.push(3), { priority: 50 });
      
      await emitter.emit({ type: 'test', timestamp: Date.now() });
      
      expect(order).toEqual([2, 3, 1]);
    });
  });

  describe('off', () => {
    it('should remove handler', () => {
      const handler = vi.fn();
      const id = emitter.on('test', handler);
      
      expect(emitter.off(id)).toBe(true);
      expect(emitter.listenerCount('test')).toBe(0);
    });
  });
});
```

## 优先级定义

| 优先级 | 说明 | 适用场景 |
|--------|------|----------|
| 100 | 最高优先级 | 日志记录、审计 |
| 50 | 高优先级 | 错误处理、状态同步 |
| 10 | 默认优先级 | 业务逻辑 |
| 1 | 低优先级 | UI更新 |

## 验收标准

- [ ] 发布/订阅模式正常工作
- [ ] 优先级排序正确，高优先级订阅者先接收事件
- [ ] 一次性订阅触发后自动取消
- [ ] 事件取消（cancelled=true）阻止后续订阅者接收
- [ ] 单事件分发延迟 < 0.1ms
- [ ] 支持异步事件处理器
- [ ] 支持AbortSignal取消订阅
