# 错误处理与恢复机制实施计划

## 模块概述

**包名**: `@ai-digital-life/errors`

**优先级**: P0 - 系统稳定性的基石

**目标**: 为应用提供统一的错误分类、传播、隔离与恢复能力。

## 设计原则

- **分类治理**: 不同类型错误采用不同处理策略
- **错误隔离**: 单个模块的错误不影响其他模块正常运行
- **自动恢复**: 可恢复错误自动重试，减少用户干预
- **可观测**: 错误信息结构化记录，支持问题回溯与统计分析

### 任务2: 重试策略 (1天)

```typescript
// src/recovery/retry.ts

import { sleep } from '@ai-digital-life/kernel';
import type { ErrorInfo } from '../types';

export type BackoffStrategy = 'linear' | 'exponential' | 'fixed';

export interface RetryOptions {
  maxRetries: number;
  initialDelay: number;
  maxDelay?: number;
  backoff: BackoffStrategy;
  jitter?: boolean;
  shouldRetry?: (error: Error, attempt: number) => boolean;
}

const DEFAULT_OPTIONS: RetryOptions = {
  maxRetries: 3,
  initialDelay: 1000,
  backoff: 'exponential',
  jitter: true,
};

export class RetryStrategy {
  private options: RetryOptions;

  constructor(options: Partial<RetryOptions> = {}) {
    this.options = { ...DEFAULT_OPTIONS, ...options };
  }

  async execute<T>(fn: () => Promise<T>): Promise<T> {
    let lastError: Error | undefined;
    
    for (let attempt = 0; attempt <= this.options.maxRetries; attempt++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error as Error;
        
        if (attempt === this.options.maxRetries) {
          break;
        }
        
        if (this.options.shouldRetry && !this.options.shouldRetry(lastError, attempt)) {
          break;
        }
        
        if (!this.isRetryable(lastError)) {
          break;
        }
        
        const delay = this.calculateDelay(attempt);
        await sleep(delay);
      }
    }
    
    throw lastError;
  }

  private calculateDelay(attempt: number): number {
    let delay: number;
    
    switch (this.options.backoff) {
      case 'exponential':
        delay = this.options.initialDelay * Math.pow(2, attempt);
        break;
      case 'linear':
        delay = this.options.initialDelay * (attempt + 1);
        break;
      case 'fixed':
      default:
        delay = this.options.initialDelay;
    }
    
    if (this.options.maxDelay) {
      delay = Math.min(delay, this.options.maxDelay);
    }
    
    if (this.options.jitter) {
      delay = delay * (0.5 + Math.random() * 0.5);
    }
    
    return Math.floor(delay);
  }

  private isRetryable(error: Error): boolean {
    if ('info' in error) {
      const info = (error as Error & { info: ErrorInfo }).info;
      return info.retryable;
    }
    
    return true;
  }
}

export function retry<T>(
  fn: () => Promise<T>,
  options?: Partial<RetryOptions>
): Promise<T> {
  const strategy = new RetryStrategy(options);
  return strategy.execute(fn);
}
```

### 任务3: 断路器 (1天)

```typescript
// src/recovery/circuit-breaker.ts

export type CircuitState = 'closed' | 'open' | 'half-open';

export interface CircuitBreakerOptions {
  failureThreshold: number;
  successThreshold: number;
  timeout: number;
  onStateChange?: (state: CircuitState) => void;
}

export class CircuitBreaker {
  private state: CircuitState = 'closed';
  private failures = 0;
  private successes = 0;
  private lastFailureTime = 0;
  private options: CircuitBreakerOptions;

  constructor(options: Partial<CircuitBreakerOptions> = {}) {
    this.options = {
      failureThreshold: options.failureThreshold ?? 5,
      successThreshold: options.successThreshold ?? 3,
      timeout: options.timeout ?? 30000,
      onStateChange: options.onStateChange,
    };
  }

  async execute<T>(fn: () => Promise<T>): Promise<T> {
    if (this.state === 'open') {
      if (Date.now() - this.lastFailureTime >= this.options.timeout) {
        this.transitionTo('half-open');
      } else {
        throw new Error('Circuit breaker is open');
      }
    }

    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  getState(): CircuitState {
    return this.state;
  }

  private onSuccess(): void {
    this.failures = 0;

    if (this.state === 'half-open') {
      this.successes++;
      if (this.successes >= this.options.successThreshold) {
        this.transitionTo('closed');
      }
    }
  }

  private onFailure(): void {
    this.failures++;
    this.successes = 0;
    this.lastFailureTime = Date.now();

    if (this.state === 'half-open') {
      this.transitionTo('open');
    } else if (this.failures >= this.options.failureThreshold) {
      this.transitionTo('open');
    }
  }

  private transitionTo(newState: CircuitState): void {
    const oldState = this.state;
    this.state = newState;

    if (newState === 'closed') {
      this.failures = 0;
      this.successes = 0;
    } else if (newState === 'open') {
      this.successes = 0;
    }

    this.options.onStateChange?.(newState);
  }

  reset(): void {
    this.transitionTo('closed');
  }
}
```

```typescript
// src/boundary/error-boundary.ts

import { createLogger } from '@ai-digital-life/logger';
import type { ErrorInfo, ErrorCategory } from '../types';

const logger = createLogger('error-boundary');

export interface ErrorHandler {
  (error: Error, context?: Record<string, unknown>): void | Promise<void>;
}

export interface ErrorBoundaryOptions {
  name: string;
  handlers?: Partial<Record<ErrorCategory, ErrorHandler>>;
  fallback?: ErrorHandler;
  rethrow?: boolean;
}

export class ErrorBoundary {
  private name: string;
  private handlers: Partial<Record<ErrorCategory, ErrorHandler>>;
  private fallback?: ErrorHandler;
  private rethrow: boolean;

  constructor(options: ErrorBoundaryOptions) {
    this.name = options.name;
    this.handlers = options.handlers ?? {};
    this.fallback = options.fallback;
    this.rethrow = options.rethrow ?? false;
  }

  async wrap<T>(fn: () => Promise<T>, context?: Record<string, unknown>): Promise<T | undefined> {
    try {
      return await fn();
    } catch (error) {
      await this.handle(error as Error, context);
      
      if (this.rethrow) {
        throw error;
      }
      
      return undefined;
    }
  }

  wrapSync<T>(fn: () => T, context?: Record<string, unknown>): T | undefined {
    try {
      return fn();
    } catch (error) {
      this.handleSync(error as Error, context);
      
      if (this.rethrow) {
        throw error;
      }
      
      return undefined;
    }
  }

  private async handle(error: Error, context?: Record<string, unknown>): Promise<void> {
    const category = this.getCategory(error);
    const handler = this.handlers[category] ?? this.fallback;

    logger.error(`Error in boundary [${this.name}]`, error, context);

    if (handler) {
      try {
        await handler(error, context);
      } catch (handlerError) {
        logger.error('Error in error handler', handlerError);
      }
    }
  }

  private handleSync(error: Error, context?: Record<string, unknown>): void {
    const category = this.getCategory(error);
    const handler = this.handlers[category] ?? this.fallback;

    logger.error(`Error in boundary [${this.name}]`, error, context);

    if (handler) {
      try {
        const result = handler(error, context);
        if (result instanceof Promise) {
          result.catch(e => logger.error('Error in async error handler', e));
        }
      } catch (handlerError) {
        logger.error('Error in error handler', handlerError);
      }
    }
  }

  private getCategory(error: Error): ErrorCategory {
    if ('info' in error) {
      return (error as Error & { info: ErrorInfo }).info.category;
    }
    return 'system' as ErrorCategory;
  }
}

export function createBoundary(options: ErrorBoundaryOptions): ErrorBoundary {
  return new ErrorBoundary(options);
}
```

```typescript
// src/recovery/fallback.ts

export interface FallbackOptions<T> {
  fallbackValue?: T;
  fallbackFn?: () => T | Promise<T>;
  logFallback?: boolean;
}

export class FallbackStrategy<T> {
  private options: FallbackOptions<T>;

  constructor(options: FallbackOptions<T> = {}) {
    this.options = {
      logFallback: true,
      ...options,
    };
  }

  async execute(fn: () => Promise<T>): Promise<T> {
    try {
      return await fn();
    } catch (error) {
      if (this.options.logFallback) {
        console.warn('Fallback triggered:', error);
      }

      if (this.options.fallbackFn) {
        return await this.options.fallbackFn();
      }

      if ('fallbackValue' in this.options) {
        return this.options.fallbackValue as T;
      }

      throw error;
    }
  }
}

export function withFallback<T>(
  fn: () => Promise<T>,
  options: FallbackOptions<T> = {}
): Promise<T> {
  const strategy = new FallbackStrategy(options);
  return strategy.execute(fn);
}
```

```typescript
import { createLogger } from '@ai-digital-life/logger';
import { sanitizeSensitiveData } from '@ai-digital-life/kernel';
import type { ErrorInfo } from '../types';

const logger = createLogger('error-reporter');

export interface ErrorReport {
  id: string;
  timestamp: number;
  error: {
    name: string;
    message: string;
    stack?: string;
    info?: ErrorInfo;
  };
  context: {
    module?: string;
    operation?: string;
    [key: string]: unknown;
  };
  system: {
    platform: string;
    version: string;
    memory: NodeJS.MemoryUsage;
  };
  userDescription?: string;
}

export class ErrorReporter {
  async generateReport(
    error: Error,
    context: Record<string, unknown> = {}
  ): Promise<ErrorReport> {
    const report: ErrorReport = {
      id: this.generateId(),
      timestamp: Date.now(),
      error: {
        name: error.name,
        message: error.message,
        stack: error.stack,
        info: 'info' in error ? (error as Error & { info: ErrorInfo }).info : undefined,
      },
      context: sanitizeSensitiveData(context) as Record<string, unknown>,
      system: {
        platform: process.platform,
        version: process.version,
        memory: process.memoryUsage(),
      },
    };

    logger.info('Error report generated', { reportId: report.id });

    return report;
  }

  async exportReport(report: ErrorReport): Promise<string> {
    return JSON.stringify(report, null, 2);
  }

  private generateId(): string {
    return `err_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }
}
```

## 验收标准

- 四类错误（系统/网络/AI/用户）分类正确，属性完整
- 错误边界隔离有效，模块内错误不导致其他模块崩溃
- 重试策略按配置执行，指数退避算法正确
- 断路器状态机转换正确（Closed→Open→HalfOpen→Closed）
- 降级方案在错误触发时正确激活
- 错误报告生成完整，脱敏处理正确

## 修订说明

错误模块不应反向依赖 `storage` 做脱敏，否则会破坏基础设施层依赖方向。

脱敏函数应放在 `kernel` 或独立的通用安全支持层中，由日志、错误报告和审计模块共同复用。

| 事件系统 | 中依赖  | 错误事件发布 |
